    85→// ─── Custom dark tooltip ─────────────────────────────────────────────────────
    86→function DarkTooltip({
    87→  active,
    88→  payload,
    89→  label,
    90→  valueSuffix = '',
    91→}: {
    92→  active?: boolean
    93→  payload?: Array<{ name: string; value: number; color: string }>
    94→  label?: string
    95→  valueSuffix?: string
    96→}) {
    97→  if (!active || !payload?.length) return null
    98→  return (
    99→    <div className="rounded-lg border border-border bg-card px-3 py-2 text-xs shadow-xl">
   100→      <p className="mb-1 font-medium text-foreground">{label}</p>
   101→      {payload.map((p, i) => (
   102→        <p key={i} className="text-muted-foreground">
   103→          <span className="inline-block mr-1.5 size-2 rounded-full" style={{ backgroundColor: p.color }} />
   104→          {p.name}: <span className="font-semibold text-foreground">{p.value}{valueSuffix}</span>
   105→        </p>
   106→      ))}
   107→    </div>
   108→  )
   109→}

   111→// ─── Mini sparkline for KPI cards ─────────────────────────────────────────────
   112→function MiniSparkline({ data, color }: { data: number[]; color: string }) {
   113→  const sparkData = data.slice(-12).map((v, i) => ({ i, v }))
   114→  if (sparkData.length < 2) return null
   115→  const min = Math.min(...sparkData.map((d) => d.v))
   116→  const max = Math.max(...sparkData.map((d) => d.v))
   117→  const range = max - min || 1
   118→  const h = 28
   119→  const w = 72
   120→  const step = w / (sparkData.length - 1)
   121→  const points = sparkData
   122→    .map((d, idx) => {
   123→      const x = idx * step
   124→      const y = h - ((d.v - min) / range) * (h - 4) - 2
   125→      return `${x},${y}`
   126→    })
   127→    .join(' ')
   128→  return (
   129→    <svg viewBox={`0 0 ${w} ${h}`} className="w-[72px] h-[28px] opacity-60">
   130→      <polyline
   131→        fill="none"
   132→        stroke={color}
   133→        strokeWidth="1.5"
   134→        strokeLinejoin="round"
   135→        strokeLinecap="round"
   136→        points={points}
   137→      />
   138→    </svg>
   139→  )
   140→}

   142→// ─── KPI Card ─────────────────────────────────────────────────────────────────
   143→interface KPICardProps {
   144→  icon: React.ElementType
   145→  label: string
   146→  value: string | number
   147→  subtitle: string
   148→  trend?: 'up' | 'down' | 'neutral'
   149→  trendValue?: string
   150→  sparkData?: number[]
   151→  sparkColor?: string
   152→  accentColor?: string
   153→  onClick?: () => void
   154→}

   156→function KPICard({
   157→  icon: Icon,
   158→  label,
   159→  value,
   160→  subtitle,
   161→  trend = 'neutral',
   162→  trendValue,
   163→  sparkData,
   164→  sparkColor = C_GREEN,
   165→  accentColor,
   166→  onClick,
   167→}: KPICardProps) {
   168→  return (
   169→    <Card
   170→      className="relative overflow-hidden py-4 transition-colors hover:border-primary/30"
   171→      onClick={onClick}
   172→      role={onClick ? 'button' : undefined}
   173→      tabIndex={onClick ? 0 : undefined}
   174→    >
   175→      {/* Subtle accent line top */}
   176→      <div
   177→        className="absolute top-0 left-0 right-0 h-0.5"
   178→        style={{ backgroundColor: accentColor || 'var(--chart-1)' }}
   179→      />
   180→      <CardContent className="flex items-start justify-between pb-0 pt-0">
   181→        <div className="flex items-start gap-3">
   182→          <div
   183→            className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg"
   184→            style={{ backgroundColor: accentColor ? `${accentColor}20` : 'var(--primary) 15%' }}
   185→          >
   186→            <Icon
   187→              className="size-5"
   188→              style={{ color: accentColor || 'var(--primary)' }}
   189→            />
   190→          </div>
   191→          <div className="min-w-0">
   192→            <p className="text-xs font-medium uppercase tracking-wider text-muted-foreground">
   193→              {label}
   194→            </p>
   195→            <p className="mt-1 text-2xl font-bold tracking-tight text-foreground">
   196→              {value}
   197→            </p>
   198→            <div className="mt-1 flex items-center gap-1.5 text-xs">
   199→              {trend === 'up' && <TrendingUp className="size-3 text-emerald-400" />}
   200→              {trend === 'down' && <TrendingDown className="size-3 text-red-400" />}
   201→              <span
   202→                className={
   203→                  trend === 'up'
   204→                    ? 'text-emerald-400'
   205→                    : trend === 'down'
   206→                      ? 'text-red-400'
   207→                      : 'text-muted-foreground'
   208→                }
   209→              >
   210→                {subtitle}
   211→              </span>
   212→            </div>
   213→          </div>
   214→        </div>
   215→        {sparkData && sparkData.length > 1 && (
   216→          <MiniSparkline data={sparkData} color={sparkColor} />
   217→        )}
   218→      </CardContent>
   219→    </Card>
   220→  )
   221→}
