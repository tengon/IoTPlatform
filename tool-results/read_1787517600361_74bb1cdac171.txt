   254→interface RolePermission {
   255→  id: string
   256→  name: string
   257→  description: string
   258→  userCount: number
   259→  color: string
   260→  bg: string
   261→  border: string
   262→  permissions: {
   263→    Dashboard: boolean
   264→    Monitoring: boolean
   265→    Analytics: boolean
   266→    Alerts: boolean
   267→    Devices: boolean
   268→    Admin: boolean
   269→  }
   270→}
   271→
   272→const permissionKeys = ['Dashboard', 'Monitoring', 'Analytics', 'Alerts', 'Devices', 'Admin'] as const
   273→
   274→const initialRoles: RolePermission[] = [
   275→  { id: 'role-admin', name: 'Admin', description: 'Full system access with all administrative privileges', userCount: 2, color: 'text-red-400', bg: 'bg-red-500/10', border: 'border-red-500/30', permissions: { Dashboard: true, Monitoring: true, Analytics: true, Alerts: true, Devices: true, Admin: true } },
   276→  { id: 'role-manager', name: 'Manager', description: 'Supervisory access with production and analytics capabilities', userCount: 3, color: 'text-amber-400', bg: 'bg-amber-500/10', border: 'border-amber-500/30', permissions: { Dashboard: true, Monitoring: true, Analytics: true, Alerts: true, Devices: true, Admin: false } },
   277→  { id: 'role-supervisor', name: 'Supervisor', description: 'Shift oversight with monitoring and alert management', userCount: 5, color: 'text-cyan-400', bg: 'bg-cyan-500/10', border: 'border-cyan-500/30', permissions: { Dashboard: true, Monitoring: true, Analytics: true, Alerts: true, Devices: false, Admin: false } },
   278→  { id: 'role-engineer', name: 'Engineer', description: 'Technical access for device management and analytics', userCount: 8, color: 'text-emerald-400', bg: 'bg-emerald-500/10', border: 'border-emerald-500/30', permissions: { Dashboard: true, Monitoring: true, Analytics: true, Alerts: true, Devices: true, Admin: false } },
   279→  { id: 'role-operator', name: 'Operator', description: 'Basic access for monitoring and dashboard viewing', userCount: 15, color: 'text-muted-foreground', bg: 'bg-primary/10', border: 'border-primary/30', permissions: { Dashboard: true, Monitoring: true, Analytics: false, Alerts: true, Devices: false, Admin: false } },
   280→]
   281→
   282→export function RolesPermissionsPage() {
   283→  const [roles, setRoles] = useState<RolePermission[]>(initialRoles)
   284→  const [editingRole, setEditingRole] = useState<RolePermission | null>(null)
   285→  const [tempPerms, setTempPerms] = useState<RolePermission['permissions'] | null>(null)
   286→  const lastUpdate = useIIoTStore((s) => s.lastUpdate)
   287→  const [lastUpdatedText, setLastUpdatedText] = useState('')
   288→
   289→  useEffect(() => {
   290→    function update() {
   291→      if (lastUpdate) {
   292→        setLastUpdatedText(formatDistanceToNow(new Date(lastUpdate), { addSuffix: true }))
   293→      }
   294→    }
   295→    update()
   296→    const interval = setInterval(update, 10000)
   297→    return () => clearInterval(interval)
   298→  }, [lastUpdate])
   299→
   300→  const openEditDialog = (role: RolePermission) => {
   301→    setEditingRole(role)
   302→    setTempPerms({ ...role.permissions })
   303→  }
   304→
   305→  const savePermissions = () => {
   306→    if (!editingRole || !tempPerms) return
   307→    setRoles(roles.map((r) => r.id === editingRole.id ? { ...r, permissions: tempPerms } : r))
   308→    setEditingRole(null)
   309→    setTempPerms(null)
   310→  }
   311→
   312→  const togglePerm = (key: keyof RolePermission['permissions']) => {
   313→    if (!tempPerms) return
   314→    setTempPerms({ ...tempPerms, [key]: !tempPerms[key] })
   315→  }
   316→
   317→  return (
   318→    <div className="space-y-6 animate-slide-up">
   319→      <PageHeader
   320→        icon={Shield}
   321→        title="Roles & Permissions"
   322→        description="Configure user roles and permissions"
   323→        lastUpdated={lastUpdatedText}
   324→      />
   325→
   326→      <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   327→        {roles.map((role, idx) => (
   328→          <Card key={role.id} className="border-border/40 hover:border-border/60 transition-colors duration-300" style={{ animationDelay: `${idx * 50}ms` }}>
   329→            <CardHeader className="pb-3 pt-5 px-5">
   330→              <div className="flex items-start justify-between">
   331→                <div className="space-y-1">
   332→                  <div className="flex items-center gap-2">
   333→                    <CardTitle className="text-base font-semibold">{role.name}</CardTitle>
   334→                    <Badge variant="outline" className={`${role.bg} ${role.color} ${role.border} text-xs`}>{role.userCount} users</Badge>
   335→                  </div>
   336→                  <CardDescription className="text-xs">{role.description}</CardDescription>
   337→                </div>
   338→                <Button variant="ghost" size="sm" className="h-8 w-8 p-0 text-muted-foreground hover:text-foreground hover:bg-muted/50" onClick={() => openEditDialog(role)}>
   339→                  <Pencil className="size-3.5" />
   340→                </Button>
   341→              </div>
   342→            </CardHeader>
   343→            <CardContent className="px-5 pb-5">
   344→              <Separator className="mb-3" />
   345→              <div className="grid grid-cols-2 gap-x-4 gap-y-2.5">
   346→                {permissionKeys.map((key) => (
   347→                  <div key={key} className="flex items-center gap-2">
   348→                    <div className={`size-4 rounded flex items-center justify-center shrink-0 ${role.permissions[key] ? 'bg-emerald-500/20' : 'bg-muted/50'}`}>
   349→                      {role.permissions[key] && (
   350→                        <svg className="size-3 text-emerald-400" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={3}>
   351→                          <path strokeLinecap="round" strokeLinejoin="round" d="M5 13l4 4L19 7" />
   352→                        </svg>
   353→                      )}
   354→                    </div>
   355→                    <span className={`text-xs ${role.permissions[key] ? 'text-foreground' : 'text-muted-foreground/60'}`}>{key}</span>
   356→                  </div>
   357→                ))}
   358→              </div>
   359→            </CardContent>
   360→          </Card>
   361→        ))}
   362→      </div>
   363→
   364→      <Dialog open={!!editingRole} onOpenChange={(open) => { if (!open) { setEditingRole(null); setTempPerms(null) } }}>
   365→        <DialogContent className="sm:max-w-[440px]">
   366→          <DialogHeader>
   367→            <DialogTitle>Edit Permissions - {editingRole?.name}</DialogTitle>
   368→            <DialogDescription>Toggle permissions for the {editingRole?.name} role</DialogDescription>
   369→          </DialogHeader>
   370→          <Separator />
   371→          <div className="py-2 space-y-3">
   372→            {permissionKeys.map((key) => (
   373→              <div key={key} className="flex items-center justify-between py-1">
   374→                <Label className="text-sm">{key}</Label>
   375→                <Switch
   376→                  checked={tempPerms?.[key] ?? false}
   377→                  onCheckedChange={() => togglePerm(key)}
   378→                />
   379→              </div>
   380→            ))}
   381→          </div>
   382→          <Separator />
   383→          <DialogFooter>
   384→            <Button variant="outline" onClick={() => { setEditingRole(null); setTempPerms(null) }}>Cancel</Button>
   385→            <Button onClick={savePermissions}>Save Permissions</Button>
   386→          </DialogFooter>
   387→        </DialogContent>
   388→      </Dialog>
   389→    </div>
   390→  )
   391→}
