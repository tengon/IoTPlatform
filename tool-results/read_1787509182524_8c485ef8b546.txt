    65→// ─── Dark tooltip ───────────────────────────────────────────────────────────
    66→function DarkTooltip({
    67→  active,
    68→  payload,
    69→  label,
    70→}: {
    71→  active?: boolean
    72→  payload?: Array<{ name: string; value: number; color: string }>

    78→      <p className="mb-1 font-medium text-foreground">{label}</p>
    79→      {payload.map((p, i) => (
    80→        <p key={i} className="text-muted-foreground">
    81→          <span

    92→// ─── SVG Semi-circle Gauge ──────────────────────────────────────────────────
    93→function OEEGauge({ value }: { value: number }) {
    94→  const clamped = Math.max(0, Math.min(100, value))
    95→  const color = clamped >= 85 ? C_GREEN : clamped >= 70 ? C_YELLOW : C_RED
    96→  const radius = 80
    97→  const cx = 100
    98→  const cy = 95
    99→  // Semi-circle: from 180° to 0° (left to right)
   100→  const circumference = Math.PI * radius // half circle
   101→  const filled = (clamped / 100) * circumference
   102→
   103→  // Build SVG arc path
   104→  const describeArc = (
   105→    x: number,
   106→    y: number,
   107→    r: number,
   108→    startAngle: number,
   109→    endAngle: number
   110→  ) => {
   111→    const start = {
   112→      x: x + r * Math.cos((startAngle * Math.PI) / 180),
   113→      y: y + r * Math.sin((startAngle * Math.PI) / 180),
   114→    }
   115→    const end = {
   116→      x: x + r * Math.cos((endAngle * Math.PI) / 180),
   117→      y: y + r * Math.sin((endAngle * Math.PI) / 180),
   118→    }
   119→    const largeArcFlag = endAngle - startAngle <= 180 ? '0' : '1'
   120→    return `M ${start.x} ${start.y} A ${r} ${r} 0 ${largeArcFlag} 1 ${end.x} ${end.y}`
   121→  }
   122→
   123→  const bgPath = describeArc(cx, cy, radius, 180, 0)
   124→  const valueAngle = 180 - (clamped / 100) * 180
   125→  const valuePath = describeArc(cx, cy, radius, 180, valueAngle)
   126→
   127→  return (
   128→    <div className="flex flex-col items-center">
   129→      <svg width="200" height="120" viewBox="0 0 200 120">
   130→        {/* Background arc */}
   131→        <path
   132→          d={bgPath}
   133→          fill="none"
   134→          stroke="rgba(255,255,255,0.08)"
   135→          strokeWidth="14"
   136→          strokeLinecap="round"
   137→        />
   138→        {/* Value arc */}
   139→        <path
   140→          d={valuePath}
   141→          fill="none"
   142→          stroke={color}
   143→          strokeWidth="14"
   144→          strokeLinecap="round"
   145→          style={{
   146→            filter: `drop-shadow(0 0 6px ${color}60)`,
   147→          }}
   148→        />
   149→        {/* Tick marks */}
   150→        {[0, 25, 50, 75, 100].map((tick) => {
   151→          const angle = 180 - (tick / 100) * 180
   152→          const rad = (angle * Math.PI) / 180
   153→          const innerR = radius - 22
   154→          const outerR = radius - 16
   155→          return (
   156→            <line
   157→              key={tick}
   158→              x1={cx + innerR * Math.cos(rad)}
   159→              y1={cy + innerR * Math.sin(rad)}
   160→              x2={cx + outerR * Math.cos(rad)}
   161→              y2={cy + outerR * Math.sin(rad)}
   162→              stroke="rgba(255,255,255,0.2)"
   163→              strokeWidth="1.5"
   164→            />
   165→          )
   166→        })}
   167→        {/* Tick labels */}
   168→        {[0, 50, 100].map((tick) => {
   169→          const angle = 180 - (tick / 100) * 180
   170→          const rad = (angle * Math.PI) / 180
   171→          const labelR = radius - 32
   172→          return (
   173→            <text
   174→              key={tick}
   175→              x={cx + labelR * Math.cos(rad)}
   176→              y={cy + labelR * Math.sin(rad) + 3}
   177→              textAnchor="middle"
   178→              fill="#a1a1aa"
   179→              fontSize="9"
   180→            >
   181→              {tick}%
   182→            </text>
   183→          )
   184→        })}
   185→        {/* Center value */}
   186→        <text x={cx} y={cy - 15} textAnchor="middle" fill={color} fontSize="32" fontWeight="bold">
   187→          {clamped.toFixed(1)}%
   188→        </text>
   189→        <text x={cx} y={cy + 5} textAnchor="middle" fill="#a1a1aa" fontSize="11">
   190→          Overall OEE
   191→        </text>
   192→      </svg>
   193→    </div>
   194→  )
   195→}
