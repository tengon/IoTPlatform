   134→   134→// ─── SVG Semi-circle Gauge (enhanced with gradient + glow) ──────────────────
   135→   135→function OEEGauge({ value }: { value: number }) {
   136→   136→  const clamped = Math.max(0, Math.min(100, value))
   137→   137→  const color = clamped >= 85 ? C_GREEN : clamped >= 70 ? C_YELLOW : C_RED
   138→   138→  const radius = 80
   139→   139→  const cx = 100
   140→   140→  const cy = 95
   141→   141→  // Semi-circle: from 180° to 0° (left to right)
   142→   142→  const circumference = Math.PI * radius // half circle
   143→   143→  const filled = (clamped / 100) * circumference
   144→   144→
   145→   145→  const gradId = `oee-gauge-grad-${color.replace('#', '')}`
   146→   146→  const glowId = `oee-gauge-glow-${color.replace('#', '')}`
   147→   147→
   148→   148→  // Build SVG arc path
   149→   149→  const describeArc = (
   150→   150→    x: number,
   151→   151→    y: number,
   152→   152→    r: number,
   153→   153→    startAngle: number,
   154→   154→    endAngle: number
   155→   155→  ) => {
   156→   156→    const start = {
   157→   157→      x: x + r * Math.cos((startAngle * Math.PI) / 180),
   158→   158→      y: y + r * Math.sin((startAngle * Math.PI) / 180),
   159→   159→    }
   160→   160→    const end = {
   161→   161→      x: x + r * Math.cos((endAngle * Math.PI) / 180),
   162→   162→      y: y + r * Math.sin((endAngle * Math.PI) / 180),
   163→   163→    }
   164→   164→    const largeArcFlag = endAngle - startAngle <= 180 ? '0' : '1'
   165→   165→    return `M ${start.x} ${start.y} A ${r} ${r} 0 ${largeArcFlag} 1 ${end.x} ${end.y}`
   166→   166→  }
   167→   167→
   168→   168→  const bgPath = describeArc(cx, cy, radius, 180, 0)
   169→   169→  const valueAngle = 180 - (clamped / 100) * 180
   170→   170→  const valuePath = describeArc(cx, cy, radius, 180, valueAngle)
   171→   171→
   172→   172→  return (
   173→   173→    <div className="flex flex-col items-center">
   174→   174→      <svg width="200" height="120" viewBox="0 0 200 120">
   175→   175→        <defs>
   176→   176→          <linearGradient id={gradId} x1="0%" y1="0%" x2="100%" y2="0%">
   177→   177→            <stop offset="0%" stopColor={color} stopOpacity="0.6" />
   178→   178→            <stop offset="50%" stopColor={color} stopOpacity="1" />
   179→   179→            <stop offset="100%" stopColor={color} stopOpacity="0.7" />
   180→   180→          </linearGradient>
   181→   181→          <filter id={glowId}>
   182→   182→            <feGaussianBlur stdDeviation="4" result="blur" />
   183→   183→            <feFlood floodColor={color} floodOpacity="0.4" result="color" />
   184→   184→            <feComposite in="color" in2="blur" operator="in" result="glow" />
   185→   185→            <feMerge>
   186→   186→              <feMergeNode in="glow" />
   187→   187→              <feMergeNode in="SourceGraphic" />
   188→   188→            </feMerge>
   189→   189→          </filter>
   190→   190→        </defs>
   191→   191→        {/* Background arc */}
   192→   192→        <path
   193→   193→          d={bgPath}
   194→   194→          fill="none"
   195→   195→          stroke="rgba(255,255,255,0.06)"
   196→   196→          strokeWidth="14"
   197→   197→          strokeLinecap="round"
   198→   198→        />
   199→   199→        {/* Value arc with gradient + glow */}
   200→   200→        <path
   201→   201→          d={valuePath}
   202→   202→          fill="none"
   203→   203→          stroke={`url(#${gradId})`}
   204→   204→          strokeWidth="14"
   205→   205→          strokeLinecap="round"
   206→   206→          filter={`url(#${glowId})`}
   207→   207→        />
   208→   208→        {/* Tick marks */}
   209→   209→        {[0, 25, 50, 75, 100].map((tick) => {
   210→   210→          const angle = 180 - (tick / 100) * 180
   211→   211→          const rad = (angle * Math.PI) / 180
   212→   212→          const innerR = radius - 22
   213→   213→          const outerR = radius - 16
   214→   214→          return (
   215→   215→            <line
   216→   216→              key={tick}
   217→   217→              x1={cx + innerR * Math.cos(rad)}
   218→   218→              y1={cy + innerR * Math.sin(rad)}
   219→   219→              x2={cx + outerR * Math.cos(rad)}
   220→   220→              y2={cy + outerR * Math.sin(rad)}
   221→   221→              stroke="rgba(255,255,255,0.2)"
   222→   222→              strokeWidth="1.5"
   223→   223→            />
   224→   224→          )
   225→   225→        })}
   226→   226→        {/* Tick labels */}
   227→   227→        {[0, 50, 100].map((tick) => {
   228→   228→          const angle = 180 - (tick / 100) * 180
   229→   229→          const rad = (angle * Math.PI) / 180
   230→   230→          const labelR = radius - 32
   231→   231→          return (
   232→   232→            <text
   233→   233→              key={tick}
   234→   234→              x={cx + labelR * Math.cos(rad)}
   235→   235→              y={cy + labelR * Math.sin(rad) + 3}
   236→   236→              textAnchor="middle"
   237→   237→              fill="rgba(255,255,255,0.35)"
   238→   238→              fontSize="9"
   239→   239→            >
   240→   240→              {tick}%
   241→   241→            </text>
   242→   242→          )
   243→   243→        })}
   244→   244→        {/* Center value */}
   245→   245→        <text x={cx} y={cy - 15} textAnchor="middle" fill={color} fontSize="32" fontWeight="bold">
   246→   246→          {clamped.toFixed(1)}%
   247→   247→        </text>
   248→   248→        <text x={cx} y={cy + 5} textAnchor="middle" fill="rgba(255,255,255,0.4)" fontSize="11">
   249→   249→          Overall OEE
   250→   250→        </text>
   251→   251→      </svg>
   252→   252→    </div>
   253→   253→  )
   254→   254→}

   300→   300→export function OEEPage() {
   301→   301→  const { machines, lastUpdate } = useIIoTStore()
   302→   302→  const [lastUpdatedText, setLastUpdatedText] = useState('—')
   303→   303→
   304→   304→  useEffect(() => {
   305→   305→    function update() {
   306→   306→      if (lastUpdate) {
   307→   307→        setLastUpdatedText(formatDistanceToNow(new Date(lastUpdate), { addSuffix: true }))
   308→   308→      }
   309→   309→    }
   310→   310→    update()
   311→   311→    const interval = setInterval(update, 10000)
   312→   312→    return () => clearInterval(interval)
   313→   313→  }, [lastUpdate])
   314→   314→
   315→   315→  const machineData = useMemo(
   316→   316→    () => (machines.length > 0 ? machines : DEFAULT_MACHINES),
   317→   317→    [machines]
   318→   318→  )
   319→   319→
   320→   320→  // ── Overall OEE ───────────────────────────────────────────────────────────
   321→   321→  const displayOEE = useMemo(
   322→   322→    () => machineData.reduce((sum, m) => sum + m.oee, 0) / machineData.length,
   323→   323→    [machineData]
   324→   324→  )
   325→   325→
   326→   326→  const overallA = useMemo(
   327→   327→    () => machineData.reduce((sum, m) => sum + m.availability, 0) / machineData.length,
   328→   328→    [machineData]
   329→   329→  )
   330→   330→
   331→   331→  const overallP = useMemo(
   332→   332→    () => machineData.reduce((sum, m) => sum + m.performance, 0) / machineData.length,
   333→   333→    [machineData]
   334→   334→  )
   335→   335→
   336→   336→  const overallQ = useMemo(
   337→   337→    () => machineData.reduce((sum, m) => sum + m.quality, 0) / machineData.length,
   338→   338→    [machineData]
   339→   339→  )
   340→   340→
   341→   341→  // ── 24h OEE trend data ──────────────────────────────────────────────────
   342→   342→  const oeeTrend24h = useMemo(() => {
   343→   343→    const data: Array<{ hour: string; OEE: number }> = []
   344→   344→    let val = displayOEE
   345→   345→    for (let i = 0; i < 24; i++) {
   346→   346→      val = Math.max(65, Math.min(95, val + (Math.random() - 0.45) * 6))
   347→   347→      data.push({
   348→   348→        hour: `${String(i).padStart(2, '0')}:00`,
   349→   349→        OEE: Math.round(val * 10) / 10,
   350→   350→      })
   351→   351→    }
   352→   352→    return data
   353→   353→  }, [displayOEE])
   354→   354→
   355→   355→  // ── 30-day OEE trend data ─────────────────────────────────────────────────
   356→   356→  const trendData = useMemo(() => {
   357→   357→    const data: Array<{
   358→   358→      day: string
   359→   359→      Availability: number
   360→   360→      Performance: number
   361→   361→      Quality: number
   362→   362→    }> = []
   363→   363→    for (let i = 29; i >= 0; i--) {
   364→   364→      data.push({
   365→   365→        day: `D-${i}`,
   366→   366→        Availability: Math.min(100, overallA + (Math.random() - 0.5) * 6),
   367→   367→        Performance: Math.min(100, overallP + (Math.random() - 0.5) * 4),

   369→   369→      })
   370→   370→    }
   371→   371→    return data
   372→   372→  }, [overallA, overallP, overallQ])
   373→   373→
   374→   374→  // ── Loss analysis data ────────────────────────────────────────────────────
   375→   375→  const lossData = useMemo(
   376→   376→    () =>
   377→   377→      machineData.map((m) => ({
   378→   378→        name: m.name.replace(/ #\d+/, ''),
   379→   379→        'Avail. Loss': Math.round((100 - m.availability) * 10) / 10,
   380→   380→        'Perf. Loss': Math.round((100 - m.performance) * 10) / 10,
   381→   381→        'Quality Loss': Math.round((100 - m.quality) * 10) / 10,
   382→   382→      })),
   383→   383→    [machineData]
   384→   384→  )
   385→   385→

   396→   396→
   397→   397→  // ── Loss analysis table data (6 machines) ─────────────────────────────────
   398→   398→  const lossTableData = useMemo(() => {
   399→   399→    if (machines.length > 0) {
   400→   400→      return machines.slice(0, 6).map((m) => {
   401→   401→        const aLoss = Math.round((100 - m.availability) * 10) / 10
   402→   402→        const pLoss = Math.round((100 - m.performance) * 10) / 10
   403→   403→        const qLoss = Math.round((100 - m.quality) * 10) / 10
   404→   404→        let biggestLoss = 'Availability'
   405→   405→        if (pLoss >= aLoss && pLoss >= qLoss) biggestLoss = 'Performance'
   406→   406→        else if (qLoss >= aLoss && qLoss >= pLoss) biggestLoss = 'Quality'
   407→   407→        return { ...m, aLoss, pLoss, qLoss, biggestLoss }
   408→   408→      })
   409→   409→    }
   410→   410→    return FALLBACK_LOSS_MACHINES.map((m) => {
   411→   411→      const aLoss = Math.round((100 - m.availability) * 10) / 10
   412→   412→      const pLoss = Math.round((100 - m.performance) * 10) / 10
   413→   413→      const qLoss = Math.round((100 - m.quality) * 10) / 10
   414→   414→      let biggestLoss = 'Availability'
   415→   415→      if (pLoss >= aLoss && pLoss >= qLoss) biggestLoss = 'Performance'
   416→   416→      else if (qLoss >= aLoss && qLoss >= pLoss) biggestLoss = 'Quality'
   417→   417→      return { ...m, aLoss, pLoss, qLoss, biggestLoss }
   418→   418→    })
   419→   419→  }, [machines])
   420→   420→
   421→   421→  // ── Overall loss proportions for stacked bar ──────────────────────────────
   422→   422→  const overallLoss = useMemo(() => {
   423→   423→    const n = lossTableData.length
   424→   424→    const aLoss = lossTableData.reduce((s, m) => s + m.aLoss, 0) / n
   425→   425→    const pLoss = lossTableData.reduce((s, m) => s + m.pLoss, 0) / n
   426→   426→    const qLoss = lossTableData.reduce((s, m) => s + m.qLoss, 0) / n
   427→   427→    const total = aLoss + pLoss + qLoss
   428→   428→    return { aLoss, pLoss, qLoss, total, aPct: (aLoss / total) * 100, pPct: (pLoss / total) * 100, qPct: (qLoss / total) * 100 }
   429→   429→  }, [lossTableData])
   430→   430→
   431→   431→  return (
   432→   432→    <div className="space-y-6">
   433→   433→      <PageHeader
   434→   434→        icon={Gauge}
   435→   435→        title="OEE Analysis"
   436→   436→        description="Overall Equipment Effectiveness breakdown"
   437→   437→        lastUpdated={lastUpdatedText}
   438→   438→      />
   439→   439→
