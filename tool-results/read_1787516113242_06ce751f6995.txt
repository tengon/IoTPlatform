   114→   114→// ─── Enhanced Mini sparkline for KPI cards ───────────────────────────────────
   115→   115→function MiniSparkline({ data, color }: { data: number[]; color: string }) {
   116→   116→  const sparkData = data.slice(-12).map((v, i) => ({ i, v }))
   117→   117→  if (sparkData.length < 2) return null
   118→   118→  const min = Math.min(...sparkData.map((d) => d.v))
   119→   119→  const max = Math.max(...sparkData.map((d) => d.v))
   120→   120→  const range = max - min || 1
   121→   121→  const h = 32
   122→   122→  const w = 80
   123→   123→  const step = w / (sparkData.length - 1)
   124→   124→  const points = sparkData
   125→   125→    .map((d, idx) => {
   126→   126→      const x = idx * step
   127→   127→      const y = h - ((d.v - min) / range) * (h - 6) - 3
   128→   128→      return `${x},${y}`
   129→   129→    })
   130→   130→    .join(' ')
   131→   131→  const lastX = (sparkData.length - 1) * step
   132→   132→  const lastY = h - ((sparkData[sparkData.length - 1].v - min) / range) * (h - 6) - 3
   133→   133→  const gradId = `spark-${color.replace('#', '')}`
   134→   134→  return (
   135→   135→    <svg viewBox={`0 0 ${w} ${h}`} className="w-20 h-8 opacity-70">
   136→   136→      <defs>
   137→   137→        <linearGradient id={gradId} x1="0" y1="0" x2="0" y2="1">
   138→   138→          <stop offset="0%" stopColor={color} stopOpacity="0.2" />
   139→   139→          <stop offset="100%" stopColor={color} stopOpacity="0.02" />
   140→   140→        </linearGradient>
   141→   141→      </defs>
   142→   142→      <polygon
   143→   143→        fill={`url(#${gradId})`}
   144→   144→        points={`0,${h} ${points} ${lastX},${h}`}
   145→   145→      />
   146→   146→      <polyline
   147→   147→        fill="none"
   148→   148→        stroke={color}
   149→   149→        strokeWidth="1.5"
   150→   150→        strokeLinejoin="round"
   151→   151→        strokeLinecap="round"
   152→   152→        points={points}
   153→   153→      />
   154→   154→      <circle cx={lastX} cy={lastY} r="2.5" fill={color} />
   155→   155→    </svg>
   156→   156→  )
   157→   157→}

   300→   300→

   302→   302→  useEffect(() => {
   303→   303→    function update() {
   304→   304→      if (lastUpdate) {

   313→   313→  // ── Computed values ──
   314→   314→  const onlineDevices = useMemo(
   315→   315→    () => devices.filter((d) => d.status === 'online' || d.status === 'warning').length,
   316→   316→    [devices]
   317→   317→  )

   319→   319→  const activeMachines = useMemo(
   320→   320→    () => machines.filter((m) => m.status === 'running').length,
   321→   321→    [machines]
   322→   322→  )
   323→   323→  const activeAlarms = useMemo(
   324→   324→    () => alarms.filter((a) => a.status === 'active'),

   335→   335→
   336→   336→  // ── Production trend data (simulated 24h) ──
   337→   337→  const productionTrendData = useMemo(() => {
   338→   338→    const hours: { hour: string; output: number; target: number }[] = []
   339→   339→    const totalTarget = production.reduce((s, p) => s + p.target, 0) || 1
   340→   340→    const totalActual = production.reduce((s, p) => s + p.actual, 0) || 0
   341→   341→    for (let h = 0; h < 24; h++) {
   342→   342→      const hourStr = `${String(h).padStart(2, '0')}:00`
   343→   343→      const shiftMultiplier = h >= 6 && h <= 18 ? 1 : 0.4
   344→   344→      const peakMultiplier = h >= 9 && h <= 16 ? 1.2 : 1
   345→   345→      const noise = 0.85 + Math.sin(h * 0.7) * 0.15
   346→   346→      const output = Math.round(
   347→   347→        (totalActual / 24) * shiftMultiplier * peakMultiplier * noise
   348→   348→      )
   349→   349→      const target = Math.round((totalTarget / 24) * shiftMultiplier * peakMultiplier)
   350→   350→      hours.push({ hour: hourStr, output, target })
   351→   351→    }
   352→   352→    return hours
   353→   353→  }, [production])
