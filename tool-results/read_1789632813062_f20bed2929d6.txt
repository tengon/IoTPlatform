   131→// ─── Enhanced Mini sparkline for KPI cards ───────────────────────────────────
   132→function MiniSparkline({ data, color }: { data: number[]; color: string }) {
   133→  const sparkData = data.slice(-12).map((v, i) => ({ i, v }))
   134→  if (sparkData.length < 2) return null
   135→  const min = Math.min(...sparkData.map((d) => d.v))
   136→  const max = Math.max(...sparkData.map((d) => d.v))
   137→  const range = max - min || 1
   138→  const h = 32
   139→  const w = 80
   140→  const step = w / (sparkData.length - 1)
   141→  const points = sparkData
   142→    .map((d, idx) => {
   143→      const x = idx * step
   144→      const y = h - ((d.v - min) / range) * (h - 6) - 3
   145→      return `${x},${y}`
   146→    })
   147→    .join(' ')
   148→  const lastX = (sparkData.length - 1) * step
   149→  const lastY = h - ((sparkData[sparkData.length - 1].v - min) / range) * (h - 6) - 3
   150→  const gradId = `spark-${color.replace('#', '')}`
   151→  return (
   152→    <svg viewBox={`0 0 ${w} ${h}`} className="w-20 h-8 opacity-70">
   153→      <defs>
   154→        <linearGradient id={gradId} x1="0" y1="0" x2="0" y2="1">
   155→          <stop offset="0%" stopColor={color} stopOpacity="0.2" />
   156→          <stop offset="100%" stopColor={color} stopOpacity="0.02" />
   157→        </linearGradient>
   158→      </defs>
   159→      <polygon
   160→        fill={`url(#${gradId})`}
   161→        points={`0,${h} ${points} ${lastX},${h}`}
   162→      />
   163→      <polyline
   164→        fill="none"
   165→        stroke={color}
   166→        strokeWidth="1.5"
   167→        strokeLinejoin="round"
   168→        strokeLinecap="round"
   169→        points={points}
   170→      />
   171→      <circle cx={lastX} cy={lastY} r="2.5" fill={color} />
   172→    </svg>
   173→  )
   174→}

   310→// ─── Main Dashboard Page ──────────────────────────────────────────────────────
   311→export function DashboardPage() {

   313→  const { data: serverMachinesRaw = [] } = useMachinesQuery()
   314→  const { data: serverDevicesRaw = [] } = useDevicesQuery()
   315→  const { data: serverAlarmsRaw = [] } = useAlarmsQuery()
   316→  const { data: serverProductionRaw = [] } = useProductionQuery()
   317→  const { data: serverEnergyRaw = [] } = useEnergyQuery()
   318→
   319→  // Normalize server data to match WS shape (add missing fields)
   320→  const serverMachines = useMemo(() =>
   321→    serverMachinesRaw.map((m: any) => ({
   322→      ...m,
   323→      status: m.status || 'running',
   324→      healthScore: m.healthScore ?? Math.floor(Math.random() * 29) + 70,
   325→    })),
   326→    [serverMachinesRaw]
   327→  )
   328→  const serverDevices = useMemo(() =>
   329→    serverDevicesRaw.map((d: any) => ({
   330→      ...d,
   331→      lastSeen: d.lastSeen || new Date().toISOString(),
   332→    })),
   333→    [serverDevicesRaw]
   334→  )
   335→  const serverAlarms = useMemo(() =>
   336→    serverAlarmsRaw.map((a: any) => ({
   337→      ...a,
   338→      alarmId: a.alarmId || a.id,
   339→      createdAt: a.createdAt || new Date().toISOString(),
   340→    })),
   341→    [serverAlarmsRaw]
   342→  )
   343→  const serverProduction = useMemo(() =>
   344→    serverProductionRaw.map((p: any) => ({
   345→      ...p,
   346→      startTime: p.startTime || new Date().toISOString(),
   347→    })),
   348→    [serverProductionRaw]
   349→  )
   350→  const serverEnergy = useMemo(() =>
   351→    (serverEnergyRaw as any[]).map((e: any) => ({
   352→      timestamp: e.timestamp,
   353→      kwh: e.kwh ?? 0,
   354→      voltage: e.voltage ?? 0,
   355→      current: e.current ?? 0,
   356→      powerFactor: e.powerFactor ?? 0,
   357→    })),
   358→    [serverEnergyRaw]
   359→  )
   360→
   361→  // ── Real-time State (WebSocket — live updates when connected) ──
   362→  const { devices: wsDevices, machines: wsMachines, alarms: wsAlarms, energyHistory: wsEnergy, production: wsProduction, lastUpdate, isConnected } = useIIoTStore()
   363→
   364→  // ── Merge: prefer WS data when connected, fall back to REST data ──
   365→  const devices = isConnected && wsDevices.length > 0 ? wsDevices : serverDevices
   366→  const machines = isConnected && wsMachines.length > 0 ? wsMachines : serverMachines
   367→  const alarms = isConnected && wsAlarms.length > 0 ? wsAlarms : serverAlarms
   368→  const production = isConnected && wsProduction.length > 0 ? wsProduction : serverProduction
   369→  const energyHistory = isConnected && wsEnergy.length > 0 ? wsEnergy : serverEnergy
   370→
   371→  const { setCurrentPage } = useNavigation()
   372→  const [selectedMachine, setSelectedMachine] = useState<MachineStatus | null>(null)
   373→  const [machineDialogOpen, setMachineDialogOpen] = useState(false)
   374→  const [lastUpdatedText, setLastUpdatedText] = useState('—')
   375→
   376→  // Effective lastUpdate: use WS heartbeat when available, otherwise use current time for REST data
   377→  const effectiveLastUpdate = lastUpdate || Date.now()
   378→
   379→  // Update last-updated text
   380→  useEffect(() => {
   381→    function update() {
   382→      if (effectiveLastUpdate) {
   383→        setLastUpdatedText(formatDistanceToNow(new Date(effectiveLastUpdate), { addSuffix: true }))
   384→      }
   385→    }
   386→    update()
   387→    const interval = setInterval(update, 10000)
   388→    return () => clearInterval(interval)
   389→  }, [effectiveLastUpdate])
   390→
   391→  // ── Computed values ──
   392→  const onlineDevices = useMemo(
   393→    () => devices.filter((d) => d.status === 'online' || d.status === 'warning').length,
   394→    [devices]
   395→  )
   396→  const offlineDevices = devices.length - onlineDevices
   397→  const activeMachines = useMemo(
   398→    () => machines.filter((m) => m.status === 'running').length,
   399→    [machines]
   400→  )
   401→  const activeAlarms = useMemo(
   402→    () => alarms.filter((a) => a.status === 'active'),
   403→    [alarms]
   404→  )
   405→  const criticalAlarms = useMemo(
   406→    () => activeAlarms.filter((a) => a.severity === 'critical').length,
   407→    [activeAlarms]
   408→  )
   409→  const todayEnergy = useMemo(() => {
   410→    if (!energyHistory.length) return 0
   411→    return energyHistory.reduce((sum, e) => sum + e.kwh, 0)
   412→  }, [energyHistory])
   413→
   414→  // ── Production trend data (simulated 24h) ──
   415→  const productionTrendData = useMemo(() => {
   416→    const hours: { hour: string; output: number; target: number }[] = []
   417→    const totalTarget = production.reduce((s, p) => s + p.target, 0) || 1
   418→    const totalActual = production.reduce((s, p) => s + p.actual, 0) || 0
   419→    for (let h = 0; h < 24; h++) {
   420→      const hourStr = `${String(h).padStart(2, '0')}:00`
   421→      const shiftMultiplier = h >= 6 && h <= 18 ? 1 : 0.4
   422→      const peakMultiplier = h >= 9 && h <= 16 ? 1.2 : 1
   423→      const noise = 0.85 + Math.sin(h * 0.7) * 0.15
   424→      const output = Math.round(
   425→        (totalActual / 24) * shiftMultiplier * peakMultiplier * noise
   426→      )
   427→      const target = Math.round((totalTarget / 24) * shiftMultiplier * peakMultiplier)
   428→      hours.push({ hour: hourStr, output, target })
   429→    }

   431→  }, [production])

   453→      { name: 'Info', value: counts.info, fill: C_CYAN },
   454→    ].filter((d) => d.value > 0)
   455→  }, [activeAlarms])
   456→
   457→  // ── Latest 5 active alarms ──

   460→      [...activeAlarms]

   463→    [activeAlarms]

   465→
   466→  // ── Energy chart data ──

   471→      return {
   472→        time: `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`,
   473→        kWh: Number(e.kwh.toFixed(2)),
   474→        voltage: Number(e.voltage.toFixed(1)),
   475→      }
   476→    })
   477→  }, [energyHistory])
   478→
   479→  // ── Sparkline data for KPIs ──
   480→  const deviceSparkData = useMemo(() => {
   481→    return [devices.length - 1, devices.length, devices.length + 1, devices.length, onlineDevices, onlineDevices, onlineDevices + 1]
   482→  }, [devices.length, onlineDevices])
   483→
   484→  const energySparkData = useMemo(
   485→    () => energyHistory.slice(-12).map((e) => e.kwh),
   486→    [energyHistory]
   487→  )
   488→
   489→  const machineSparkData = useMemo(
   490→    () => machines.map((m) => (m.status === 'running' ? 1 : 0)),
   491→    [machines]
   492→  )
   493→
   494→  const staggerClass = (i: number) => `stagger-${Math.min(i + 1, 6)}`

   497→    <div className="flex flex-col gap-6">
   498→      <PageHeader
   499→        title="Dashboard"
   500→        description="Real-time industrial operations overview"
   501→        icon={Activity}
   502→        lastUpdated={lastUpdatedText}
   503→        badge={isConnected ? 'LIVE' : undefined}
   504→        actions={
   505→          <div className="flex items-center gap-2">
   506→            <Button variant="outline" size="sm" className="h-8 text-xs gap-1.5">
   507→              <Calendar className="size-3.5" />

   511→              <Download className="size-3.5" />

   516→      />
   517→
   518→      {/* ── KPI Cards Row ── */}
