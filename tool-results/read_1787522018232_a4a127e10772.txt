   256→interface RolePermission {
   257→  id: string
   258→  name: string
   259→  description: string
   260→  userCount: number
   261→  color: string
   262→  bg: string
   263→  border: string
   264→  permissions: {
   265→    Dashboard: boolean
   266→    Monitoring: boolean
   267→    Analytics: boolean
   268→    Alerts: boolean
   269→    Devices: boolean
   270→    Admin: boolean
   271→  }
   272→}
   273→
   274→const permissionKeys = ['Dashboard', 'Monitoring', 'Analytics', 'Alerts', 'Devices', 'Admin'] as const
   275→
   276→const initialRoles: RolePermission[] = [
   277→  { id: 'role-admin', name: 'Admin', description: 'Full system access with all administrative privileges', userCount: 2, color: 'text-red-400', bg: 'bg-red-500/10', border: 'border-red-500/30', permissions: { Dashboard: true, Monitoring: true, Analytics: true, Alerts: true, Devices: true, Admin: true } },

   279→  { id: 'role-supervisor', name: 'Supervisor', description: 'Shift oversight with monitoring and alert management', userCount: 5, color: 'text-cyan-400', bg: 'bg-cyan-500/10', border: 'border-cyan-500/30', permissions: { Dashboard: true, Monitoring: true, Analytics: true, Alerts: true, Devices: false, Admin: false } },

   282→]
   283→
   284→export function RolesPermissionsPage() {
   285→  const [roles, setRoles] = useState<RolePermission[]>(initialRoles)
   286→  const [editingRole, setEditingRole] = useState<RolePermission | null>(null)
   287→  const [tempPerms, setTempPerms] = useState<RolePermission['permissions'] | null>(null)
   288→  const lastUpdate = useIIoTStore((s) => s.lastUpdate)
   289→  const [lastUpdatedText, setLastUpdatedText] = useState('')
   290→
   291→  useEffect(() => {
   292→    function update() {
   293→      if (lastUpdate) {
   294→        setLastUpdatedText(formatDistanceToNow(new Date(lastUpdate), { addSuffix: true }))
   295→      }
   296→    }
   297→    update()
   298→    const interval = setInterval(update, 10000)
   299→    return () => clearInterval(interval)
   300→  }, [lastUpdate])
   301→
   302→  const openEditDialog = (role: RolePermission) => {
   303→    setEditingRole(role)
   304→    setTempPerms({ ...role.permissions })
   305→  }
   306→
   307→  const savePermissions = () => {
   308→    if (!editingRole || !tempPerms) return
   309→    setRoles(roles.map((r) => r.id === editingRole.id ? { ...r, permissions: tempPerms } : r))
   310→    setEditingRole(null)
   311→    setTempPerms(null)
   312→  }
   313→
   314→  const togglePerm = (key: keyof RolePermission['permissions']) => {
   315→    if (!tempPerms) return
   316→    setTempPerms({ ...tempPerms, [key]: !tempPerms[key] })
   317→  }
   318→
   319→  return (
   320→    <div className="space-y-6 animate-slide-up">
   321→      <PageHeader
   322→        icon={Shield}
   323→        title="Roles & Permissions"
   324→        description="Configure user roles and permissions"
   325→        lastUpdated={lastUpdatedText}
   326→      />
   327→
   328→      <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   329→        {roles.map((role, idx) => (
   330→          <Card key={role.id} className="border-border/40 hover:border-border/60 transition-colors duration-300" style={{ animationDelay: `${idx * 50}ms` }}>
   331→            <CardHeader className="pb-3 pt-5 px-5">
   332→              <div className="flex items-start justify-between">
   333→                <div className="space-y-1">
   334→                  <div className="flex items-center gap-2">
   335→                    <CardTitle className="text-base font-semibold">{role.name}</CardTitle>
   336→                    <Badge variant="outline" className={`${role.bg} ${role.color} ${role.border} text-xs`}>{role.userCount} users</Badge>
   337→                  </div>
   338→                  <CardDescription className="text-xs">{role.description}</CardDescription>
   339→                </div>
   340→                <Button variant="ghost" size="sm" className="h-8 w-8 p-0 text-muted-foreground hover:text-foreground hover:bg-muted/50" onClick={() => openEditDialog(role)}>
   341→                  <Pencil className="size-3.5" />
   342→                </Button>
   343→              </div>
   344→            </CardHeader>
   345→            <CardContent className="px-5 pb-5">
   346→              <Separator className="mb-3" />
   347→              <div className="grid grid-cols-2 gap-x-4 gap-y-2.5">
   348→                {permissionKeys.map((key) => (
   349→                  <div key={key} className="flex items-center gap-2">
   350→                    <div className={`size-4 rounded flex items-center justify-center shrink-0 ${role.permissions[key] ? 'bg-emerald-500/20' : 'bg-muted/50'}`}>
   351→                      {role.permissions[key] && (
   352→                        <svg className="size-3 text-emerald-400" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={3}>
   353→                          <path strokeLinecap="round" strokeLinejoin="round" d="M5 13l4 4L19 7" />
   354→                        </svg>
   355→                      )}
   356→                    </div>
   357→                    <span className={`text-xs ${role.permissions[key] ? 'text-foreground' : 'text-muted-foreground/60'}`}>{key}</span>
   358→                  </div>
   359→                ))}
   360→              </div>
   361→            </CardContent>
   362→          </Card>
   363→        ))}
   364→      </div>
   365→
   366→      <Dialog open={!!editingRole} onOpenChange={(open) => { if (!open) { setEditingRole(null); setTempPerms(null) } }}>
   367→        <DialogContent className="sm:max-w-[440px]">
   368→          <DialogHeader>
   369→            <DialogTitle>Edit Permissions - {editingRole?.name}</DialogTitle>
   370→            <DialogDescription>Toggle permissions for the {editingRole?.name} role</DialogDescription>
   371→          </DialogHeader>
   372→          <Separator />
   373→          <div className="py-2 space-y-3">
   374→            {permissionKeys.map((key) => (
   375→              <div key={key} className="flex items-center justify-between py-1">
   376→                <Label className="text-sm">{key}</Label>
   377→                <Switch
   378→                  checked={tempPerms?.[key] ?? false}
   379→                  onCheckedChange={() => togglePerm(key)}
   380→                />
   381→              </div>
   382→            ))}
   383→          </div>
   384→          <Separator />
   385→          <DialogFooter>
   386→            <Button variant="outline" onClick={() => { setEditingRole(null); setTempPerms(null) }}>Cancel</Button>
   387→            <Button onClick={savePermissions}>Save Permissions</Button>
   388→          </DialogFooter>
   389→        </DialogContent>
   390→      </Dialog>
   391→    </div>
   392→  )
   393→}
