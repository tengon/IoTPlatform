   131→   131→// ─── Enhanced Mini sparkline for KPI cards ───────────────────────────────────
   132→   132→function MiniSparkline({ data, color }: { data: number[]; color: string }) {
   133→   133→  const sparkData = data.slice(-12).map((v, i) => ({ i, v }))
   134→   134→  if (sparkData.length < 2) return null
   135→   135→  const min = Math.min(...sparkData.map((d) => d.v))
   136→   136→  const max = Math.max(...sparkData.map((d) => d.v))
   137→   137→  const range = max - min || 1
   138→   138→  const h = 32
   139→   139→  const w = 80
   140→   140→  const step = w / (sparkData.length - 1)
   141→   141→  const points = sparkData
   142→   142→    .map((d, idx) => {
   143→   143→      const x = idx * step
   144→   144→      const y = h - ((d.v - min) / range) * (h - 6) - 3
   145→   145→      return `${x},${y}`
   146→   146→    })
   147→   147→    .join(' ')
   148→   148→  const lastX = (sparkData.length - 1) * step
   149→   149→  const lastY = h - ((sparkData[sparkData.length - 1].v - min) / range) * (h - 6) - 3
   150→   150→  const gradId = `spark-${color.replace('#', '')}`
   151→   151→  return (
   152→   152→    <svg viewBox={`0 0 ${w} ${h}`} className="w-20 h-8 opacity-70">
   153→   153→      <defs>
   154→   154→        <linearGradient id={gradId} x1="0" y1="0" x2="0" y2="1">
   155→   155→          <stop offset="0%" stopColor={color} stopOpacity="0.2" />
   156→   156→          <stop offset="100%" stopColor={color} stopOpacity="0.02" />
   157→   157→        </linearGradient>
   158→   158→      </defs>
   159→   159→      <polygon
   160→   160→        fill={`url(#${gradId})`}
   161→   161→        points={`0,${h} ${points} ${lastX},${h}`}
   162→   162→      />
   163→   163→      <polyline
   164→   164→        fill="none"
   165→   165→        stroke={color}
   166→   166→        strokeWidth="1.5"
   167→   167→        strokeLinejoin="round"
   168→   168→        strokeLinecap="round"
   169→   169→        points={points}
   170→   170→      />
   171→   171→      <circle cx={lastX} cy={lastY} r="2.5" fill={color} />
   172→   172→    </svg>
   173→   173→  )
   174→   174→}

   336→   336→    serverAlarmsRaw.map((a: any) => ({

   339→   339→      createdAt: a.createdAt || new Date().toISOString(),
   340→   340→    })),
   341→   341→    [serverAlarmsRaw]

   344→   344→    serverProductionRaw.map((p: any) => ({
   345→   345→      ...p,
   346→   346→      startTime: p.startTime || new Date().toISOString(),
   347→   347→    })),
   348→   348→    [serverProductionRaw]
   349→   349→  )
   350→   350→  const serverEnergy = useMemo(() =>
   351→   351→    (serverEnergyRaw as any[]).map((e: any) => ({
   352→   352→      timestamp: e.timestamp,
   353→   353→      kwh: e.kwh ?? 0,
   354→   354→      voltage: e.voltage ?? 0,
   355→   355→      current: e.current ?? 0,
   356→   356→      powerFactor: e.powerFactor ?? 0,
   357→   357→    })),
   358→   358→    [serverEnergyRaw]
   359→   359→  )
   360→   360→
   361→   361→  // ── Real-time State (WebSocket — live updates when connected) ──
   362→   362→  const { devices: wsDevices, machines: wsMachines, alarms: wsAlarms, energyHistory: wsEnergy, production: wsProduction, lastUpdate, isConnected } = useIIoTStore()
   363→   363→
   364→   364→  // ── Merge: prefer WS data when connected, fall back to REST data ──
   365→   365→  const devices = isConnected && wsDevices.length > 0 ? wsDevices : serverDevices
   366→   366→  const machines = isConnected && wsMachines.length > 0 ? wsMachines : serverMachines
   367→   367→  const alarms = isConnected && wsAlarms.length > 0 ? wsAlarms : serverAlarms
   368→   368→  const production = isConnected && wsProduction.length > 0 ? wsProduction : serverProduction
   369→   369→  const energyHistory = isConnected && wsEnergy.length > 0 ? wsEnergy : serverEnergy

   372→   372→  const [selectedMachine, setSelectedMachine] = useState<MachineStatus | null>(null)
   373→   373→  const [machineDialogOpen, setMachineDialogOpen] = useState(false)
   374→   374→  const [lastUpdatedText, setLastUpdatedText] = useState('—')
   375→   375→
   376→   376→  // Effective lastUpdate: use WS heartbeat when available, otherwise use current time for REST data
   377→   377→  const effectiveLastUpdate = lastUpdate || Date.now()
   378→   378→
   379→   379→  // Update last-updated text
   380→   380→  useEffect(() => {
   381→   381→    function update() {
   382→   382→      if (effectiveLastUpdate) {
   383→   383→        setLastUpdatedText(formatDistanceToNow(new Date(effectiveLastUpdate), { addSuffix: true }))
   384→   384→      }
   385→   385→    }
   386→   386→    update()
   387→   387→    const interval = setInterval(update, 10000)
   388→   388→    return () => clearInterval(interval)
   389→   389→  }, [effectiveLastUpdate])
   390→   390→
   391→   391→  // ── Computed values ──
   392→   392→  const onlineDevices = useMemo(
   393→   393→    () => devices.filter((d) => d.status === 'online' || d.status === 'warning').length,
   394→   394→    [devices]
   395→   395→  )
   396→   396→  const offlineDevices = devices.length - onlineDevices
   397→   397→  const activeMachines = useMemo(
   398→   398→    () => machines.filter((m) => m.status === 'running').length,
   399→   399→    [machines]
   400→   400→  )
   401→   401→  const activeAlarms = useMemo(
   402→   402→    () => alarms.filter((a) => a.status === 'active'),
   403→   403→    [alarms]
   404→   404→  )
   405→   405→  const criticalAlarms = useMemo(
   406→   406→    () => activeAlarms.filter((a) => a.severity === 'critical').length,
   407→   407→    [activeAlarms]
   408→   408→  )
   409→   409→  const todayEnergy = useMemo(() => {
   410→   410→    if (!energyHistory.length) return 0
   411→   411→    return energyHistory.reduce((sum, e) => sum + e.kwh, 0)
   412→   412→  }, [energyHistory])
   413→   413→
   414→   414→  // ── Production trend data (simulated 24h) ──
   415→   415→  const productionTrendData = useMemo(() => {
   416→   416→    const hours: { hour: string; output: number; target: number }[] = []
   417→   417→    const totalTarget = production.reduce((s, p) => s + p.target, 0) || 1
   418→   418→    const totalActual = production.reduce((s, p) => s + p.actual, 0) || 0
   419→   419→    for (let h = 0; h < 24; h++) {
   420→   420→      const hourStr = `${String(h).padStart(2, '0')}:00`
   421→   421→      const shiftMultiplier = h >= 6 && h <= 18 ? 1 : 0.4
   422→   422→      const peakMultiplier = h >= 9 && h <= 16 ? 1.2 : 1
   423→   423→      const noise = 0.85 + Math.sin(h * 0.7) * 0.15
   424→   424→      const output = Math.round(
   425→   425→        (totalActual / 24) * shiftMultiplier * peakMultiplier * noise
   426→   426→      )
   427→   427→      const target = Math.round((totalTarget / 24) * shiftMultiplier * peakMultiplier)
   428→   428→      hours.push({ hour: hourStr, output, target })
   429→   429→    }
   430→   430→    return hours
   431→   431→  }, [production])
   432→   432→
   433→   433→  // ── OEE Bar Chart data ──
   434→   434→  const oeeData = useMemo(
   435→   435→    () =>
   436→   436→      machines.map((m) => ({
   437→   437→        name: m.name.length > 14 ? m.name.slice(0, 14) + '…' : m.name,
   438→   438→        OEE: Number((m.oee * 100).toFixed(1)),
   439→   439→        Availability: Number((m.availability * 100).toFixed(1)),
   440→   440→        Performance: Number((m.performance * 100).toFixed(1)),
   441→   441→        Quality: Number((m.quality * 100).toFixed(1)),
   442→   442→      })),
   443→   443→    [machines]
   444→   444→  )
   445→   445→
