   114→// ─── Enhanced Mini sparkline for KPI cards ───────────────────────────────────
   115→function MiniSparkline({ data, color }: { data: number[]; color: string }) {
   116→  const sparkData = data.slice(-12).map((v, i) => ({ i, v }))
   117→  if (sparkData.length < 2) return null
   118→  const min = Math.min(...sparkData.map((d) => d.v))
   119→  const max = Math.max(...sparkData.map((d) => d.v))
   120→  const range = max - min || 1
   121→  const h = 32
   122→  const w = 80
   123→  const step = w / (sparkData.length - 1)
   124→  const points = sparkData
   125→    .map((d, idx) => {
   126→      const x = idx * step
   127→      const y = h - ((d.v - min) / range) * (h - 6) - 3
   128→      return `${x},${y}`
   129→    })
   130→    .join(' ')
   131→  const lastX = (sparkData.length - 1) * step
   132→  const lastY = h - ((sparkData[sparkData.length - 1].v - min) / range) * (h - 6) - 3
   133→  const gradId = `spark-${color.replace('#', '')}`
   134→  return (
   135→    <svg viewBox={`0 0 ${w} ${h}`} className="w-20 h-8 opacity-70">
   136→      <defs>
   137→        <linearGradient id={gradId} x1="0" y1="0" x2="0" y2="1">
   138→          <stop offset="0%" stopColor={color} stopOpacity="0.2" />
   139→          <stop offset="100%" stopColor={color} stopOpacity="0.02" />
   140→        </linearGradient>
   141→      </defs>
   142→      <polygon
   143→        fill={`url(#${gradId})`}
   144→        points={`0,${h} ${points} ${lastX},${h}`}
   145→      />
   146→      <polyline
   147→        fill="none"
   148→        stroke={color}
   149→        strokeWidth="1.5"
   150→        strokeLinejoin="round"
   151→        strokeLinecap="round"
   152→        points={points}
   153→      />
   154→      <circle cx={lastX} cy={lastY} r="2.5" fill={color} />
   155→    </svg>
   156→  )
   157→}

   247→          </div>
   248→        )}

   300→
   301→  // Update last-updated text
   302→  useEffect(() => {
   303→    function update() {
   304→      if (lastUpdate) {
   305→        setLastUpdatedText(formatDistanceToNow(new Date(lastUpdate), { addSuffix: true }))
   306→      }
   307→    }
   308→    update()
   309→    const interval = setInterval(update, 10000)
   310→    return () => clearInterval(interval)
   311→  }, [lastUpdate])
   312→
   313→  // ── Computed values ──
   314→  const onlineDevices = useMemo(
   315→    () => devices.filter((d) => d.status === 'online' || d.status === 'warning').length,
   316→    [devices]
   317→  )
   318→  const offlineDevices = devices.length - onlineDevices
   319→  const activeMachines = useMemo(
   320→    () => machines.filter((m) => m.status === 'running').length,
   321→    [machines]
   322→  )
   323→  const activeAlarms = useMemo(
   324→    () => alarms.filter((a) => a.status === 'active'),
   325→    [alarms]
   326→  )
   327→  const criticalAlarms = useMemo(
   328→    () => activeAlarms.filter((a) => a.severity === 'critical').length,
   329→    [activeAlarms]
   330→  )
   331→  const todayEnergy = useMemo(() => {
   332→    if (!energyHistory.length) return 0
   333→    return energyHistory.reduce((sum, e) => sum + e.kwh, 0)
   334→  }, [energyHistory])
   335→
   336→  // ── Production trend data (simulated 24h) ──
   337→  const productionTrendData = useMemo(() => {
   338→    const hours: { hour: string; output: number; target: number }[] = []
   339→    const totalTarget = production.reduce((s, p) => s + p.target, 0) || 1
   340→    const totalActual = production.reduce((s, p) => s + p.actual, 0) || 0
   341→    for (let h = 0; h < 24; h++) {
   342→      const hourStr = `${String(h).padStart(2, '0')}:00`
   343→      const shiftMultiplier = h >= 6 && h <= 18 ? 1 : 0.4
   344→      const peakMultiplier = h >= 9 && h <= 16 ? 1.2 : 1
   345→      const noise = 0.85 + Math.sin(h * 0.7) * 0.15
   346→      const output = Math.round(
   347→        (totalActual / 24) * shiftMultiplier * peakMultiplier * noise
   348→      )
   349→      const target = Math.round((totalTarget / 24) * shiftMultiplier * peakMultiplier)
   350→      hours.push({ hour: hourStr, output, target })
   351→    }
   352→    return hours
   353→  }, [production])
   354→
   355→  // ── OEE Bar Chart data ──
   356→  const oeeData = useMemo(
   357→    () =>
   358→      machines.map((m) => ({
   359→        name: m.name.length > 14 ? m.name.slice(0, 14) + '…' : m.name,
   360→        OEE: Number((m.oee * 100).toFixed(1)),
   361→        Availability: Number((m.availability * 100).toFixed(1)),
   362→        Performance: Number((m.performance * 100).toFixed(1)),
   363→        Quality: Number((m.quality * 100).toFixed(1)),
   364→      })),
   365→    [machines]
   366→  )
   367→
   368→  // ── Alarm distribution ──
   369→  const alarmDistribution = useMemo(() => {
   370→    const counts = { critical: 0, warning: 0, info: 0 }
   371→    activeAlarms.forEach((a) => { counts[a.severity]++ })
   372→    return [
   373→      { name: 'Critical', value: counts.critical, fill: C_RED },
   374→      { name: 'Warning', value: counts.warning, fill: C_YELLOW },
   375→      { name: 'Info', value: counts.info, fill: C_CYAN },
   376→    ].filter((d) => d.value > 0)
   377→  }, [activeAlarms])
   378→
   379→  // ── Latest 5 active alarms ──
   380→  const latestAlarms = useMemo(
   381→    () =>
   382→      [...activeAlarms]
   383→        .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   384→        .slice(0, 5),
   385→    [activeAlarms]
   386→  )
   387→
   388→  // ── Energy chart data ──
   389→  const energyChartData = useMemo(() => {
   390→    if (!energyHistory.length) return []

   392→      const d = new Date(e.timestamp)
   393→      return {
   394→        time: `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`,
   395→        kWh: Number(e.kwh.toFixed(2)),
   396→        voltage: Number(e.voltage.toFixed(1)),
   397→      }
   398→    })
   399→  }, [energyHistory])
   400→
   401→  // ── Sparkline data for KPIs ──
   402→  const deviceSparkData = useMemo(() => {
   403→    return [devices.length - 1, devices.length, devices.length + 1, devices.length, onlineDevices, onlineDevices, onlineDevices + 1]

   405→
   406→  const energySparkData = useMemo(
   407→    () => energyHistory.slice(-12).map((e) => e.kwh),
   408→    [energyHistory]
   409→  )
   410→
   411→  const machineSparkData = useMemo(
   412→    () => machines.map((m) => (m.status === 'running' ? 1 : 0)),
   413→    [machines]

   415→

   423→        icon={Activity}
   424→        lastUpdated={lastUpdatedText}
   425→        badge={isConnected ? 'LIVE' : undefined}
   426→        actions={
   427→          <div className="flex items-center gap-2">
   428→            <Button variant="outline" size="sm" className="h-8 text-xs gap-1.5">
   429→              <Calendar className="size-3.5" />
