   133→// ─── SVG Semi-circle Gauge (enhanced with gradient + glow) ──────────────────
   134→function OEEGauge({ value }: { value: number }) {
   135→  const clamped = Math.max(0, Math.min(100, value))
   136→  const color = clamped >= 85 ? C_GREEN : clamped >= 70 ? C_YELLOW : C_RED
   137→  const radius = 80
   138→  const cx = 100
   139→  const cy = 95
   140→  // Semi-circle: from 180° to 0° (left to right)
   141→  const circumference = Math.PI * radius // half circle
   142→  const filled = (clamped / 100) * circumference
   143→
   144→  const gradId = `oee-gauge-grad-${color.replace('#', '')}`
   145→  const glowId = `oee-gauge-glow-${color.replace('#', '')}`
   146→
   147→  // Build SVG arc path
   148→  const describeArc = (
   149→    x: number,
   150→    y: number,
   151→    r: number,
   152→    startAngle: number,
   153→    endAngle: number
   154→  ) => {
   155→    const start = {
   156→      x: x + r * Math.cos((startAngle * Math.PI) / 180),
   157→      y: y + r * Math.sin((startAngle * Math.PI) / 180),
   158→    }
   159→    const end = {
   160→      x: x + r * Math.cos((endAngle * Math.PI) / 180),
   161→      y: y + r * Math.sin((endAngle * Math.PI) / 180),
   162→    }
   163→    const largeArcFlag = endAngle - startAngle <= 180 ? '0' : '1'
   164→    return `M ${start.x} ${start.y} A ${r} ${r} 0 ${largeArcFlag} 1 ${end.x} ${end.y}`
   165→  }
   166→
   167→  const bgPath = describeArc(cx, cy, radius, 180, 0)
   168→  const valueAngle = 180 - (clamped / 100) * 180
   169→  const valuePath = describeArc(cx, cy, radius, 180, valueAngle)
   170→
   171→  return (
   172→    <div className="flex flex-col items-center">
   173→      <svg width="200" height="120" viewBox="0 0 200 120">
   174→        <defs>
   175→          <linearGradient id={gradId} x1="0%" y1="0%" x2="100%" y2="0%">
   176→            <stop offset="0%" stopColor={color} stopOpacity="0.6" />
   177→            <stop offset="50%" stopColor={color} stopOpacity="1" />
   178→            <stop offset="100%" stopColor={color} stopOpacity="0.7" />
   179→          </linearGradient>
   180→          <filter id={glowId}>
   181→            <feGaussianBlur stdDeviation="4" result="blur" />
   182→            <feFlood floodColor={color} floodOpacity="0.4" result="color" />
   183→            <feComposite in="color" in2="blur" operator="in" result="glow" />
   184→            <feMerge>
   185→              <feMergeNode in="glow" />
   186→              <feMergeNode in="SourceGraphic" />
   187→            </feMerge>
   188→          </filter>
   189→        </defs>
   190→        {/* Background arc */}
   191→        <path
   192→          d={bgPath}
   193→          fill="none"
   194→          stroke="rgba(255,255,255,0.06)"
   195→          strokeWidth="14"
   196→          strokeLinecap="round"
   197→        />
   198→        {/* Value arc with gradient + glow */}
   199→        <path
   200→          d={valuePath}
   201→          fill="none"
   202→          stroke={`url(#${gradId})`}
   203→          strokeWidth="14"
   204→          strokeLinecap="round"
   205→          filter={`url(#${glowId})`}
   206→        />
   207→        {/* Tick marks */}
   208→        {[0, 25, 50, 75, 100].map((tick) => {
   209→          const angle = 180 - (tick / 100) * 180
   210→          const rad = (angle * Math.PI) / 180
   211→          const innerR = radius - 22
   212→          const outerR = radius - 16
   213→          return (
   214→            <line
   215→              key={tick}
   216→              x1={cx + innerR * Math.cos(rad)}
   217→              y1={cy + innerR * Math.sin(rad)}
   218→              x2={cx + outerR * Math.cos(rad)}
   219→              y2={cy + outerR * Math.sin(rad)}
   220→              stroke="rgba(255,255,255,0.2)"
   221→              strokeWidth="1.5"
   222→            />
   223→          )
   224→        })}
   225→        {/* Tick labels */}
   226→        {[0, 50, 100].map((tick) => {
   227→          const angle = 180 - (tick / 100) * 180
   228→          const rad = (angle * Math.PI) / 180
   229→          const labelR = radius - 32
   230→          return (
   231→            <text
   232→              key={tick}
   233→              x={cx + labelR * Math.cos(rad)}
   234→              y={cy + labelR * Math.sin(rad) + 3}
   235→              textAnchor="middle"
   236→              fill="rgba(255,255,255,0.35)"
   237→              fontSize="9"
   238→            >
   239→              {tick}%
   240→            </text>
   241→          )
   242→        })}
   243→        {/* Center value */}
   244→        <text x={cx} y={cy - 15} textAnchor="middle" fill={color} fontSize="32" fontWeight="bold">
   245→          {clamped.toFixed(1)}%
   246→        </text>
   247→        <text x={cx} y={cy + 5} textAnchor="middle" fill="rgba(255,255,255,0.4)" fontSize="11">
   248→          Overall OEE
   249→        </text>
   250→      </svg>
   251→    </div>
   252→  )
   253→}

   298→// ─── Component ──────────────────────────────────────────────────────────────
   299→export function OEEPage() {
   300→  const { machines, lastUpdate } = useIIoTStore()
   301→  const [lastUpdatedText, setLastUpdatedText] = useState('—')
   302→
   303→  useEffect(() => {
   304→    function update() {
   305→      if (lastUpdate) {
   306→        setLastUpdatedText(formatDistanceToNow(new Date(lastUpdate), { addSuffix: true }))
   307→      }
   308→    }
   309→    update()
   310→    const interval = setInterval(update, 10000)
   311→    return () => clearInterval(interval)
   312→  }, [lastUpdate])
   313→
   314→  const machineData = useMemo(
   315→    () => (machines.length > 0 ? machines : DEFAULT_MACHINES),
   316→    [machines]
   317→  )
   318→
   319→  // ── Overall OEE ───────────────────────────────────────────────────────────
   320→  const displayOEE = useMemo(
   321→    () => machineData.reduce((sum, m) => sum + m.oee, 0) / machineData.length,
   322→    [machineData]
   323→  )
   324→
   325→  const overallA = useMemo(
   326→    () => machineData.reduce((sum, m) => sum + m.availability, 0) / machineData.length,
   327→    [machineData]
   328→  )
   329→
   330→  const overallP = useMemo(
   331→    () => machineData.reduce((sum, m) => sum + m.performance, 0) / machineData.length,
   332→    [machineData]
   333→  )
   334→
   335→  const overallQ = useMemo(
   336→    () => machineData.reduce((sum, m) => sum + m.quality, 0) / machineData.length,
   337→    [machineData]
   338→  )
   339→
   340→  // ── 24h OEE trend data ──────────────────────────────────────────────────
   341→  const oeeTrend24h = useMemo(() => {
   342→    const data: Array<{ hour: string; OEE: number }> = []
   343→    let val = displayOEE
   344→    for (let i = 0; i < 24; i++) {
   345→      val = Math.max(65, Math.min(95, val + (Math.random() - 0.45) * 6))
   346→      data.push({
   347→        hour: `${String(i).padStart(2, '0')}:00`,
   348→        OEE: Math.round(val * 10) / 10,
   349→      })
   350→    }
   351→    return data
   352→  }, [displayOEE])
   353→
   354→  // ── 30-day OEE trend data ─────────────────────────────────────────────────
   355→  const trendData = useMemo(() => {
   356→    const data: Array<{
   357→      day: string
   358→      Availability: number
   359→      Performance: number
   360→      Quality: number
   361→    }> = []
   362→    for (let i = 29; i >= 0; i--) {
   363→      data.push({
   364→        day: `D-${i}`,
   365→        Availability: Math.min(100, overallA + (Math.random() - 0.5) * 6),
   366→        Performance: Math.min(100, overallP + (Math.random() - 0.5) * 4),
   367→        Quality: Math.min(100, overallQ + (Math.random() - 0.5) * 1.5),
   368→      })
   369→    }
   370→    return data
   371→  }, [overallA, overallP, overallQ])
   372→
   373→  // ── Loss analysis data ────────────────────────────────────────────────────
   374→  const lossData = useMemo(
   375→    () =>
   376→      machineData.map((m) => ({
   377→        name: m.name.replace(/ #\d+/, ''),
   378→        'Avail. Loss': Math.round((100 - m.availability) * 10) / 10,
   379→        'Perf. Loss': Math.round((100 - m.performance) * 10) / 10,
   380→        'Quality Loss': Math.round((100 - m.quality) * 10) / 10,
   381→      })),
   382→    [machineData]
   383→  )
   384→
   385→  // ── Target vs Actual ──────────────────────────────────────────────────────
   386→  const targetVsActual = useMemo(
   387→    () =>
   388→      machineData.map((m) => ({
   389→        name: m.name.replace(/ #\d+/, ''),
   390→        Target: OEE_TARGET,
   391→        Actual: m.oee,
   392→      })),
   393→    [machineData]
   394→  )
   395→
   396→  // ── Loss analysis table data (6 machines) ─────────────────────────────────
   397→  const lossTableData = useMemo(() => {
   398→    if (machines.length > 0) {
   399→      return machines.slice(0, 6).map((m) => {
   400→        const aLoss = Math.round((100 - m.availability) * 10) / 10
   401→        const pLoss = Math.round((100 - m.performance) * 10) / 10
   402→        const qLoss = Math.round((100 - m.quality) * 10) / 10
   403→        let biggestLoss = 'Availability'
   404→        if (pLoss >= aLoss && pLoss >= qLoss) biggestLoss = 'Performance'
   405→        else if (qLoss >= aLoss && qLoss >= pLoss) biggestLoss = 'Quality'
   406→        return { ...m, aLoss, pLoss, qLoss, biggestLoss }
   407→      })
   408→    }
   409→    return FALLBACK_LOSS_MACHINES.map((m) => {
   410→      const aLoss = Math.round((100 - m.availability) * 10) / 10
   411→      const pLoss = Math.round((100 - m.performance) * 10) / 10
   412→      const qLoss = Math.round((100 - m.quality) * 10) / 10
   413→      let biggestLoss = 'Availability'
   414→      if (pLoss >= aLoss && pLoss >= qLoss) biggestLoss = 'Performance'
   415→      else if (qLoss >= aLoss && qLoss >= pLoss) biggestLoss = 'Quality'
   416→      return { ...m, aLoss, pLoss, qLoss, biggestLoss }
   417→    })
   418→  }, [machines])
   419→
   420→  // ── Overall loss proportions for stacked bar ──────────────────────────────
   421→  const overallLoss = useMemo(() => {
   422→    const n = lossTableData.length
   423→    const aLoss = lossTableData.reduce((s, m) => s + m.aLoss, 0) / n
   424→    const pLoss = lossTableData.reduce((s, m) => s + m.pLoss, 0) / n
   425→    const qLoss = lossTableData.reduce((s, m) => s + m.qLoss, 0) / n
   426→    const total = aLoss + pLoss + qLoss
   427→    return { aLoss, pLoss, qLoss, total, aPct: (aLoss / total) * 100, pPct: (pLoss / total) * 100, qPct: (qLoss / total) * 100 }
   428→  }, [lossTableData])
   429→
   430→  return (
   431→    <div className="space-y-6">
   432→      <PageHeader
   433→        icon={Gauge}
   434→        title="OEE Analysis"
   435→        description="Overall Equipment Effectiveness breakdown"
   436→        lastUpdated={lastUpdatedText}
   437→      />
   438→
   439→      {/* OEE Gauge */}
   440→      <Card className="chart-container-glass hover:border-border/60 transition-colors duration-300 animate-slide-up">
   441→        <CardContent className="pt-6 flex flex-col items-center px-5 pb-5">
   442→          <OEEGauge value={displayOEE} />
   443→          <div className="flex items-center gap-2 mt-2">
   444→            <Target className="h-3.5 w-3.5 text-muted-foreground" />
   445→            <span className="text-xs text-muted-foreground">
   446→              Target: {OEE_TARGET}%{' '}
   447→              {displayOEE >= OEE_TARGET ? (
   448→                <span className="text-emerald-400 font-medium">
   449→                  <ArrowUpRight className="inline h-3 w-3" /> On Track
   450→                </span>
   451→              ) : (
   452→                <span className="text-red-400 font-medium">
   453→                  <ArrowDownRight className="inline h-3 w-3" /> Below Target
   454→                </span>
   455→              )}
   456→            </span>
   457→          </div>
   458→        </CardContent>
   459→      </Card>
   460→
   461→      {/* OEE Trend (24h) */}
   462→      <ChartCard
   463→        title="OEE Trend (24h)"
   464→        description="Hourly OEE percentage over the last 24 hours"
   465→        icon={TrendingUp}
   466→        className="chart-container-glass animate-slide-up stagger-1"
   467→      >
   468→        <div className="h-[250px] w-full">
   469→          <ResponsiveContainer width="100%" height="100%">
   470→            <AreaChart data={oeeTrend24h}>
   471→              <defs>
   472→                <linearGradient id="oeeTrendGrad" x1="0" y1="0" x2="0" y2="1">
   473→                  <stop offset="0%" stopColor={C_GREEN} stopOpacity={0.35} />
   474→                  <stop offset="100%" stopColor={C_GREEN} stopOpacity={0.02} />
   475→                </linearGradient>
   476→              </defs>
   477→              <CartesianGrid strokeDasharray="3 3" stroke={GRID_STROKE} />
   478→              <XAxis dataKey="hour" tick={AXIS_TICK_SM} tickLine={false} axisLine={AXIS_LINE} interval={3} />
   479→              <YAxis tick={AXIS_TICK_SM} tickLine={false} axisLine={AXIS_LINE} domain={[60, 100]} width={35} />
   480→              <Tooltip content={<ChartTooltip valueSuffix="%" />} />
   481→              <Area
   482→                type="monotone"
   483→                dataKey="OEE"
   484→                stroke={C_GREEN}
   485→                fill="url(#oeeTrendGrad)"
   486→                strokeWidth={2}
   487→                dot={false}
   488→              />
   489→            </AreaChart>
   490→          </ResponsiveContainer>
   491→        </div>
   492→      </ChartCard>
   493→
   494→      {/* A × P × Q = OEE Breakdown */}
   495→      <div className="animate-slide-up stagger-2">
   496→        <p className="text-xs text-muted-foreground mb-3 text-center">
   497→          OEE = Availability × Performance × Quality
   498→        </p>
   499→        <div className="flex flex-wrap items-center justify-center gap-2">
   500→          <OEEComponentCard
   501→            label="Availability"
   502→            value={overallA}
   503→            icon={() => <span className="text-lg font-mono" style={{ color: C_GREEN }}>A</span>}
   504→            color={C_GREEN}
   505→          />
   506→          <span className="text-2xl font-bold text-muted-foreground shrink-0">×</span>
   507→          <OEEComponentCard
   508→            label="Performance"
   509→            value={overallP}
   510→            icon={() => <span className="text-lg font-mono" style={{ color: C_CYAN }}>P</span>}
   511→            color={C_CYAN}
   512→          />
   513→          <span className="text-2xl font-bold text-muted-foreground shrink-0">×</span>
   514→          <OEEComponentCard
   515→            label="Quality"
   516→            value={overallQ}
   517→            icon={() => <span className="text-lg font-mono" style={{ color: C_ORANGE }}>Q</span>}
