   392→// ─── Generate hourly chart data ─────────────────────────────────────────
   393→function generateHourlyData() {
   394→  const data: { hour: string; events: number }[] = []
   395→  const now = new Date()
   396→  for (let i = 23; i >= 0; i--) {
   397→    const hour = subHours(now, i)
   398→    // Simulate realistic activity pattern: low at night, peaks during work hours
   399→    const h = hour.getHours()
   400→    let base: number
   401→    if (h >= 0 && h < 6) base = 2 + Math.random() * 3
   402→    else if (h >= 6 && h < 8) base = 5 + Math.random() * 6
   403→    else if (h >= 8 && h < 12) base = 10 + Math.random() * 8
   404→    else if (h >= 12 && h < 14) base = 7 + Math.random() * 5
   405→    else if (h >= 14 && h < 18) base = 10 + Math.random() * 8
   406→    else if (h >= 18 && h < 21) base = 5 + Math.random() * 4
   407→    else base = 3 + Math.random() * 3
   408→    data.push({
   409→      hour: format(hour, 'HH:00'),
   410→      events: Math.round(base),
   411→    })
   412→  }
   413→  return data
   414→}

   517→

   521→

   527→
   528→  const toggleRow = (id: string) => {

   531→      if (next.has(id)) next.delete(id)

   535→  }
   536→
   537→  const hourlyData = useMemo(() => generateHourlyData(), [])
   538→

   545→        lastUpdated={lastUpdate ? formatDistanceToNow(new Date(lastUpdate), { addSuffix: true }) : undefined}
   546→        badge={`${filtered.length} events`}
   547→        actions={
   548→          <Button variant="outline" size="sm" className="h-8 text-xs gap-1.5">
   549→            <Download className="size-3.5" />
   550→            Export Log

   552→        }
   553→      />
   554→

   556→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">

   560→          value={kpis.total}
   561→          unit="events"

   565→          index={0}
   566→        />
   567→        <KPICard
   568→          icon={UserCog}
   569→          label="User Actions"
   570→          value={kpis.userActions}
   571→          unit="events"
   572→          trend="up"
   573→          trendValue={`${Math.round((kpis.userActions / Math.max(kpis.total, 1)) * 100)}% of total`}
   574→          color={C_CYAN}
   575→          index={1}
   576→        />
   577→        <KPICard
   578→          icon={Clock}
   579→          label="System Events"
   580→          value={kpis.systemEvents}
   581→          unit="events"
   582→          trend="stable"
   583→          trendValue="Automated processes"
   584→          color={C_ORANGE}
   585→          index={2}
   586→        />
   587→        <KPICard
   588→          icon={Shield}
   589→          label="Security Events"
   590→          value={kpis.securityEvents}
   591→          unit="events"
   592→          trend={kpis.securityEvents > 3 ? 'up' : 'stable'}
   593→          trendValue={kpis.securityEvents > 3 ? 'Elevated activity' : 'Normal'}
   594→          color={C_RED}
   595→          index={3}
   596→        />
   597→      </div>
   598→
   599→      {/* ── Activity Timeline Chart ────────────────────────────────── */}
   600→      <Card className="border-border/40 stagger-2 animate-slide-up">
   601→        <CardContent className="p-4">
   602→          <div className="flex items-center justify-between mb-3">
   603→            <h3 className="text-sm font-semibold">Activity Timeline</h3>
   604→            <span className="text-[11px] text-muted-foreground/60 uppercase tracking-wider">Events per hour (24h)</span>

   612→                  tick={AXIS_TICK_SM}
   613→                  axisLine={AXIS_LINE}
   614→                  tickLine={false}

   621→                  allowDecimals={false}

   623→                <Tooltip content={<ChartTooltip />} cursor={{ fill: 'rgba(255,255,255,0.03)' }} />

   625→                  dataKey="events"
   626→                  fill={C_GREEN}
   627→                  radius={[3, 3, 0, 0]}
   628→                  fillOpacity={0.8}

   630→              </BarChart>

   633→        </CardContent>

   636→      {/* ── Filter Bar + Table ─────────────────────────────────────── */}
   637→      <Card className="border-border/40 stagger-3 animate-slide-up">
   638→        <CardContent className="p-4 space-y-4">
   639→          {/* Filter Bar */}
   640→          <div className="flex flex-col sm:flex-row sm:items-center gap-3">

   700→            <Table>
   701→              <TableHeader>
   702→                <TableRow className="hover:bg-transparent border-border/30">

   722→                    </TableCell>
   723→                  </TableRow>
   724→                ) : (
   725→                  paginatedEntries.map((entry, idx) => {
   726→                    const isExpanded = expandedRows.has(entry.id)
   727→                    const actionCfg = ACTION_CONFIG[entry.actionType]
   728→                    const catCfg = CATEGORY_CONFIG[entry.category]
   729→                    return (
   730→                      <Collapsible
   731→                        key={entry.id}

   733→                        onOpenChange={() => toggleRow(entry.id)}
   734→                      >
   735→                        <CollapsibleTrigger asChild>
   736→                          <TableRow
   737→                            className={cn(
   738→                              'cursor-pointer hover:bg-muted/20 transition-colors duration-150 table-row-interactive zebra-row',
   739→                              isExpanded && 'bg-muted/10'
   740→                            )}
   741→                          >

   744→                                ? <ChevronDown className="size-3.5 text-muted-foreground" />
   745→                                : <ChevronRight className="size-3.5 text-muted-foreground" />
   746→                              }
   747→                            </TableCell>
   748→                            <TableCell className="py-3 whitespace-nowrap">
   749→                              <span className="text-xs text-muted-foreground">
   750→                                {formatDistanceToNow(new Date(entry.timestamp), { addSuffix: true })}
   751→                              </span>
   752→                            </TableCell>
   753→                            <TableCell className="py-3">
   754→                              <span className={cn('text-sm font-medium', entry.user === 'System' ? 'text-muted-foreground/70' : 'text-foreground')}>{entry.user}</span>
   755→                            </TableCell>
   756→                            <TableCell className="py-3">
   757→                              <Badge variant="outline" className={cn('text-[10px] font-semibold px-2 py-0.5 badge-sharp', actionCfg.className)}>
   758→                                {actionCfg.label}
   759→                              </Badge>
   760→                            </TableCell>
   761→                            <TableCell className="py-3 hidden lg:table-cell">
   762→                              <Badge variant="outline" className={cn('text-[10px] px-2 py-0.5', catCfg.className)}>
   763→                                {catCfg.label}
   764→                              </Badge>
   765→                            </TableCell>
   766→                            <TableCell className="py-3 hidden md:table-cell">
   767→                              <span className="text-sm text-muted-foreground/80 max-w-[200px] truncate block">
   768→                                {entry.resource}
   769→                              </span>
   770→                            </TableCell>
   771→                            <TableCell className="py-3 hidden xl:table-cell">
   772→                              <span className="text-xs text-muted-foreground/60 max-w-[240px] truncate block">
   773→                                {entry.details}
   774→                              </span>
   775→                            </TableCell>
   776→                            <TableCell className="py-3 hidden lg:table-cell">
   777→                              <code className="text-[11px] text-muted-foreground/50 font-mono bg-muted/50 px-1.5 py-0.5 rounded">
   778→                                {entry.ipAddress}
   779→                              </code>
   780→                            </TableCell>
   781→                          </TableRow>
   782→                        </CollapsibleTrigger>
   783→                        <CollapsibleContent>
   784→                          <TableRow className="bg-muted/5 hover:bg-muted/5">
   785→                            <TableCell colSpan={8} className="py-0">
   786→                              <div className="px-4 py-3 animate-slide-up">
   787→                                <p className="text-[11px] uppercase tracking-wider text-muted-foreground/60 mb-2 font-semibold">Full Details</p>
   788→                                <pre className="text-xs text-muted-foreground/80 bg-muted/40 rounded-lg p-3 overflow-x-auto font-mono leading-relaxed border border-border/30">
   789→                                  {entry.detailsJson}
   790→                                </pre>
   791→                              </div>
   792→                            </TableCell>
   793→                          </TableRow>
   794→                        </CollapsibleContent>
   795→                      </Collapsible>
   796→                    )
   797→                  })
   798→                )}
   799→              </TableBody>
   800→            </Table>
   801→          </div>
   802→
   803→          {/* Pagination */}
   804→          <div className="flex items-center justify-between">
   805→            <p className="text-xs text-muted-foreground">
   806→              Showing {(page - 1) * perPage + 1}–{Math.min(page * perPage, filtered.length)} of {filtered.length} events
   807→            </p>
   808→            <div className="flex items-center gap-1">
   809→              <Button
   810→                variant="outline"
   811→                size="sm"
   812→                className="h-7 w-7 p-0 text-xs"
   813→                disabled={page <= 1}
   814→                onClick={() => setPage(1)}
   815→              >
   816→                «
   817→              </Button>
   818→              <Button
   819→                variant="outline"
   820→                size="sm"
   821→                className="h-7 w-7 p-0 text-xs"
   822→                disabled={page <= 1}
   823→                onClick={() => setPage((p) => p - 1)}
   824→              >
   825→                ‹
   826→              </Button>
   827→              {Array.from({ length: totalPages }, (_, i) => i + 1).map((p) => (
   828→                <Button
   829→                  key={p}
   830→                  variant={p === page ? 'default' : 'outline'}
   831→                  size="sm"
   832→                  className="h-7 w-7 p-0 text-xs"
   833→                  onClick={() => setPage(p)}
   834→                >
   835→                  {p}
   836→                </Button>
   837→              ))}
   838→              <Button
   839→                variant="outline"
   840→                size="sm"
   841→                className="h-7 w-7 p-0 text-xs"
   842→                disabled={page >= totalPages}
   843→                onClick={() => setPage((p) => p + 1)}
   844→              >
   845→                ›
   846→              </Button>
   847→              <Button
   848→                variant="outline"
   849→                size="sm"
   850→                className="h-7 w-7 p-0 text-xs"
   851→                disabled={page >= totalPages}
   852→                onClick={() => setPage(totalPages)}
   853→              >
   854→                »
   855→              </Button>
   856→            </div>
   857→          </div>
   858→        </CardContent>
   859→      </Card>
   860→    </div>
   861→  )
   862→}
