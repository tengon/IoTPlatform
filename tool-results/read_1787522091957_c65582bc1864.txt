   198→
   199→    const upcoming = tasks.filter(
   200→      (t) =>
   201→        (t.status === 'Scheduled' || t.status === 'In Progress') &&
   202→        !isBefore(new Date(t.scheduledDate), now) &&
   203→        !isBefore(weekLater, new Date(t.scheduledDate))
   204→    ).length
   205→
   206→    const overdue = tasks.filter((t) => t.status === 'Overdue').length
   207→
   208→    const completedThisMonth = tasks.filter(
   209→      (t) =>
   210→        t.status === 'Completed' &&
   211→        !isBefore(new Date(t.scheduledDate), monthStart) &&
   212→        !isBefore(monthEnd, new Date(t.scheduledDate))
   213→    ).length
   214→
   215→    const totalScheduled = tasks.filter((t) => t.status !== 'Skipped').length
   216→    const totalCompleted = tasks.filter((t) => t.status === 'Completed').length
   217→    const avgRate = totalScheduled > 0 ? ((totalCompleted / totalScheduled) * 100).toFixed(1) : '0.0'
   218→
   219→    return { upcoming, overdue, completedThisMonth, avgRate }
   220→  }, [tasks])
   221→

   223→    return tasks.filter((t) => {
   224→      if (filterStatus !== 'all' && t.status !== filterStatus) return false
   225→      if (filterPriority !== 'all' && t.priority !== filterPriority) return false
   226→      if (filterType !== 'all' && t.taskType !== filterType) return false
   227→      if (filterMachine !== 'all' && t.machine !== filterMachine) return false
   228→      return true
   229→    })
   230→  }, [tasks, filterStatus, filterPriority, filterType, filterMachine])
   231→

   235→    setFilterStatus('all')

   240→
   241→  // ─── Calendar helpers ────────────────────────────────────────────────────
   242→
   243→  const calendarDays = useMemo(() => {
   244→    const monthStart = startOfMonth(calendarMonth)
   245→    const monthEnd = endOfMonth(calendarMonth)
   246→    const days = eachDayOfInterval({ start: monthStart, end: monthEnd })
   247→    const startDay = getDay(monthStart) // 0=Sun
   248→    return { days, startDay }
   249→  }, [calendarMonth])

   253→    [tasks]

   269→      estimatedDuration: formDuration,
   270→      description: formDescription,

   277→    setFormPriority('Medium')
   278→    setFormDate(format(new Date(), 'yyyy-MM-dd'))
   279→    setFormTechnician('')
   280→    setFormDuration('')
   281→    setFormDescription('')
   282→  }, [formMachine, formDate, formTechnician, formDuration, formType, formPriority, formDescription, tasks.length])
