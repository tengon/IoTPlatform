   134→// ─── SVG Semi-circle Gauge (enhanced with gradient + glow) ──────────────────
   135→function OEEGauge({ value }: { value: number }) {
   136→  const clamped = Math.max(0, Math.min(100, value))
   137→  const color = clamped >= 85 ? C_GREEN : clamped >= 70 ? C_YELLOW : C_RED
   138→  const radius = 80
   139→  const cx = 100
   140→  const cy = 95
   141→  // Semi-circle: from 180° to 0° (left to right)
   142→  const circumference = Math.PI * radius // half circle
   143→  const filled = (clamped / 100) * circumference
   144→
   145→  const gradId = `oee-gauge-grad-${color.replace('#', '')}`
   146→  const glowId = `oee-gauge-glow-${color.replace('#', '')}`
   147→
   148→  // Build SVG arc path
   149→  const describeArc = (
   150→    x: number,
   151→    y: number,
   152→    r: number,
   153→    startAngle: number,
   154→    endAngle: number
   155→  ) => {
   156→    const start = {
   157→      x: x + r * Math.cos((startAngle * Math.PI) / 180),
   158→      y: y + r * Math.sin((startAngle * Math.PI) / 180),
   159→    }
   160→    const end = {
   161→      x: x + r * Math.cos((endAngle * Math.PI) / 180),
   162→      y: y + r * Math.sin((endAngle * Math.PI) / 180),
   163→    }
   164→    const largeArcFlag = endAngle - startAngle <= 180 ? '0' : '1'
   165→    return `M ${start.x} ${start.y} A ${r} ${r} 0 ${largeArcFlag} 1 ${end.x} ${end.y}`
   166→  }
   167→
   168→  const bgPath = describeArc(cx, cy, radius, 180, 0)
   169→  const valueAngle = 180 - (clamped / 100) * 180
   170→  const valuePath = describeArc(cx, cy, radius, 180, valueAngle)
   171→
   172→  return (
   173→    <div className="flex flex-col items-center">
   174→      <svg width="200" height="120" viewBox="0 0 200 120">
   175→        <defs>
   176→          <linearGradient id={gradId} x1="0%" y1="0%" x2="100%" y2="0%">
   177→            <stop offset="0%" stopColor={color} stopOpacity="0.6" />
   178→            <stop offset="50%" stopColor={color} stopOpacity="1" />
   179→            <stop offset="100%" stopColor={color} stopOpacity="0.7" />
   180→          </linearGradient>
   181→          <filter id={glowId}>
   182→            <feGaussianBlur stdDeviation="4" result="blur" />
   183→            <feFlood floodColor={color} floodOpacity="0.4" result="color" />
   184→            <feComposite in="color" in2="blur" operator="in" result="glow" />
   185→            <feMerge>
   186→              <feMergeNode in="glow" />
   187→              <feMergeNode in="SourceGraphic" />
   188→            </feMerge>
   189→          </filter>
   190→        </defs>
   191→        {/* Background arc */}
   192→        <path
   193→          d={bgPath}
   194→          fill="none"
   195→          stroke="rgba(255,255,255,0.06)"
   196→          strokeWidth="14"
   197→          strokeLinecap="round"
   198→        />
   199→        {/* Value arc with gradient + glow */}
   200→        <path
   201→          d={valuePath}
   202→          fill="none"
   203→          stroke={`url(#${gradId})`}
   204→          strokeWidth="14"
   205→          strokeLinecap="round"
   206→          filter={`url(#${glowId})`}
   207→        />
   208→        {/* Tick marks */}
   209→        {[0, 25, 50, 75, 100].map((tick) => {
   210→          const angle = 180 - (tick / 100) * 180
   211→          const rad = (angle * Math.PI) / 180
   212→          const innerR = radius - 22
   213→          const outerR = radius - 16
   214→          return (
   215→            <line
   216→              key={tick}
   217→              x1={cx + innerR * Math.cos(rad)}
   218→              y1={cy + innerR * Math.sin(rad)}
   219→              x2={cx + outerR * Math.cos(rad)}
   220→              y2={cy + outerR * Math.sin(rad)}
   221→              stroke="rgba(255,255,255,0.2)"
   222→              strokeWidth="1.5"
   223→            />
   224→          )
   225→        })}
   226→        {/* Tick labels */}
   227→        {[0, 50, 100].map((tick) => {
   228→          const angle = 180 - (tick / 100) * 180
   229→          const rad = (angle * Math.PI) / 180
   230→          const labelR = radius - 32
   231→          return (
   232→            <text
   233→              key={tick}
   234→              x={cx + labelR * Math.cos(rad)}
   235→              y={cy + labelR * Math.sin(rad) + 3}
   236→              textAnchor="middle"
   237→              fill="rgba(255,255,255,0.35)"
   238→              fontSize="9"
   239→            >
   240→              {tick}%
   241→            </text>
   242→          )
   243→        })}
   244→        {/* Center value */}
   245→        <text x={cx} y={cy - 15} textAnchor="middle" fill={color} fontSize="32" fontWeight="bold">
   246→          {clamped.toFixed(1)}%
   247→        </text>
   248→        <text x={cx} y={cy + 5} textAnchor="middle" fill="rgba(255,255,255,0.4)" fontSize="11">
   249→          Overall OEE
   250→        </text>
   251→      </svg>
   252→    </div>
   253→  )
   254→}

   299→// ─── Component ──────────────────────────────────────────────────────────────
   300→export function OEEPage() {
   301→  const { machines, lastUpdate } = useIIoTStore()
   302→  const [lastUpdatedText, setLastUpdatedText] = useState('—')
   303→
   304→  useEffect(() => {
   305→    function update() {
   306→      if (lastUpdate) {
   307→        setLastUpdatedText(formatDistanceToNow(new Date(lastUpdate), { addSuffix: true }))
   308→      }
   309→    }
   310→    update()
   311→    const interval = setInterval(update, 10000)
   312→    return () => clearInterval(interval)
   313→  }, [lastUpdate])
   314→
   315→  const machineData = useMemo(
   316→    () => (machines.length > 0 ? machines : DEFAULT_MACHINES),
   317→    [machines]
   318→  )
   319→
   320→  // ── Overall OEE ───────────────────────────────────────────────────────────
   321→  const displayOEE = useMemo(
   322→    () => machineData.reduce((sum, m) => sum + m.oee, 0) / machineData.length,
   323→    [machineData]
   324→  )
   325→
   326→  const overallA = useMemo(
   327→    () => machineData.reduce((sum, m) => sum + m.availability, 0) / machineData.length,
   328→    [machineData]
   329→  )
   330→
   331→  const overallP = useMemo(
   332→    () => machineData.reduce((sum, m) => sum + m.performance, 0) / machineData.length,
   333→    [machineData]
   334→  )
   335→
   336→  const overallQ = useMemo(
   337→    () => machineData.reduce((sum, m) => sum + m.quality, 0) / machineData.length,
   338→    [machineData]
   339→  )
   340→
   341→  // ── 24h OEE trend data ──────────────────────────────────────────────────
   342→  const oeeTrend24h = useMemo(() => {
   343→    const data: Array<{ hour: string; OEE: number }> = []
   344→    let val = displayOEE
   345→    for (let i = 0; i < 24; i++) {
   346→      val = Math.max(65, Math.min(95, val + (Math.random() - 0.45) * 6))
   347→      data.push({
   348→        hour: `${String(i).padStart(2, '0')}:00`,
   349→        OEE: Math.round(val * 10) / 10,
   350→      })
   351→    }
   352→    return data
   353→  }, [displayOEE])
   354→
   355→  // ── 30-day OEE trend data ─────────────────────────────────────────────────
   356→  const trendData = useMemo(() => {
   357→    const data: Array<{
   358→      day: string
   359→      Availability: number
   360→      Performance: number
   361→      Quality: number
   362→    }> = []
   363→    for (let i = 29; i >= 0; i--) {
   364→      data.push({
   365→        day: `D-${i}`,
   366→        Availability: Math.min(100, overallA + (Math.random() - 0.5) * 6),
   367→        Performance: Math.min(100, overallP + (Math.random() - 0.5) * 4),
   368→        Quality: Math.min(100, overallQ + (Math.random() - 0.5) * 1.5),
   369→      })
   370→    }
   371→    return data
   372→  }, [overallA, overallP, overallQ])
   373→
   374→  // ── Loss analysis data ────────────────────────────────────────────────────
   375→  const lossData = useMemo(
   376→    () =>
   377→      machineData.map((m) => ({
   378→        name: m.name.replace(/ #\d+/, ''),
   379→        'Avail. Loss': Math.round((100 - m.availability) * 10) / 10,
   380→        'Perf. Loss': Math.round((100 - m.performance) * 10) / 10,
   381→        'Quality Loss': Math.round((100 - m.quality) * 10) / 10,
   382→      })),
   383→    [machineData]
   384→  )
   385→
   386→  // ── Target vs Actual ──────────────────────────────────────────────────────
   387→  const targetVsActual = useMemo(
   388→    () =>
   389→      machineData.map((m) => ({
   390→        name: m.name.replace(/ #\d+/, ''),
   391→        Target: OEE_TARGET,
   392→        Actual: m.oee,
   393→      })),
   394→    [machineData]
   395→  )
   396→
   397→  // ── Loss analysis table data (6 machines) ─────────────────────────────────
   398→  const lossTableData = useMemo(() => {
   399→    if (machines.length > 0) {
   400→      return machines.slice(0, 6).map((m) => {
   401→        const aLoss = Math.round((100 - m.availability) * 10) / 10
   402→        const pLoss = Math.round((100 - m.performance) * 10) / 10
   403→        const qLoss = Math.round((100 - m.quality) * 10) / 10
   404→        let biggestLoss = 'Availability'
   405→        if (pLoss >= aLoss && pLoss >= qLoss) biggestLoss = 'Performance'
   406→        else if (qLoss >= aLoss && qLoss >= pLoss) biggestLoss = 'Quality'
   407→        return { ...m, aLoss, pLoss, qLoss, biggestLoss }
   408→      })
   409→    }
   410→    return FALLBACK_LOSS_MACHINES.map((m) => {
   411→      const aLoss = Math.round((100 - m.availability) * 10) / 10
   412→      const pLoss = Math.round((100 - m.performance) * 10) / 10
   413→      const qLoss = Math.round((100 - m.quality) * 10) / 10
   414→      let biggestLoss = 'Availability'
   415→      if (pLoss >= aLoss && pLoss >= qLoss) biggestLoss = 'Performance'
   416→      else if (qLoss >= aLoss && qLoss >= pLoss) biggestLoss = 'Quality'
   417→      return { ...m, aLoss, pLoss, qLoss, biggestLoss }
   418→    })
   419→  }, [machines])
   420→
   421→  // ── Overall loss proportions for stacked bar ──────────────────────────────
   422→  const overallLoss = useMemo(() => {
   423→    const n = lossTableData.length
   424→    const aLoss = lossTableData.reduce((s, m) => s + m.aLoss, 0) / n
   425→    const pLoss = lossTableData.reduce((s, m) => s + m.pLoss, 0) / n
   426→    const qLoss = lossTableData.reduce((s, m) => s + m.qLoss, 0) / n
   427→    const total = aLoss + pLoss + qLoss
   428→    return { aLoss, pLoss, qLoss, total, aPct: (aLoss / total) * 100, pPct: (pLoss / total) * 100, qPct: (qLoss / total) * 100 }
   429→  }, [lossTableData])
   430→
   431→  return (
   432→    <div className="space-y-6">
   433→      <PageHeader
   434→        icon={Gauge}
   435→        title="OEE Analysis"
   436→        description="Overall Equipment Effectiveness breakdown"
   437→        lastUpdated={lastUpdatedText}
   438→      />
   439→
   440→      {/* OEE Gauge */}
   441→      <Card className="chart-container-glass hover:border-border/60 transition-colors duration-300 animate-slide-up">
   442→        <CardContent className="pt-6 flex flex-col items-center px-5 pb-5">
   443→          <OEEGauge value={displayOEE} />
   444→          <div className="flex items-center gap-2 mt-2">
   445→            <Target className="h-3.5 w-3.5 text-muted-foreground" />
   446→            <span className="text-xs text-muted-foreground">
   447→              Target: {OEE_TARGET}%{' '}
   448→              {displayOEE >= OEE_TARGET ? (
   449→                <span className="text-emerald-400 font-medium">
   450→                  <ArrowUpRight className="inline h-3 w-3" /> On Track
   451→                </span>
   452→              ) : (
   453→                <span className="text-red-400 font-medium">
   454→                  <ArrowDownRight className="inline h-3 w-3" /> Below Target
   455→                </span>
   456→              )}
   457→            </span>
   458→          </div>
   459→        </CardContent>
   460→      </Card>
   461→
   462→      {/* OEE Trend (24h) */}
   463→      <ChartCard
   464→        title="OEE Trend (24h)"
   465→        description="Hourly OEE percentage over the last 24 hours"
   466→        icon={TrendingUp}
   467→        className="chart-container-glass animate-slide-up stagger-1"
   468→      >
   469→        <div className="h-[250px] w-full">
   470→          <ResponsiveContainer width="100%" height="100%">
   471→            <AreaChart data={oeeTrend24h}>
   472→              <defs>
   473→                <linearGradient id="oeeTrendGrad" x1="0" y1="0" x2="0" y2="1">
   474→                  <stop offset="0%" stopColor={C_GREEN} stopOpacity={0.35} />
   475→                  <stop offset="100%" stopColor={C_GREEN} stopOpacity={0.02} />
   476→                </linearGradient>
   477→              </defs>
   478→              <CartesianGrid strokeDasharray="3 3" stroke={GRID_STROKE} />
   479→              <XAxis dataKey="hour" tick={AXIS_TICK_SM} tickLine={false} axisLine={AXIS_LINE} interval={3} />
   480→              <YAxis tick={AXIS_TICK_SM} tickLine={false} axisLine={AXIS_LINE} domain={[60, 100]} width={35} />
   481→              <Tooltip content={<ChartTooltip valueSuffix="%" />} />
   482→              <Area
   483→                type="monotone"
   484→                dataKey="OEE"
   485→                stroke={C_GREEN}
   486→                fill="url(#oeeTrendGrad)"
   487→                strokeWidth={2}
   488→                dot={false}
   489→              />
   490→            </AreaChart>
   491→          </ResponsiveContainer>
   492→        </div>
   493→      </ChartCard>
   494→
   495→      {/* A × P × Q = OEE Breakdown */}
   496→      <div className="animate-slide-up stagger-2">
   497→        <p className="text-xs text-muted-foreground mb-3 text-center">
   498→          OEE = Availability × Performance × Quality
   499→        </p>
   500→        <div className="flex flex-wrap items-center justify-center gap-2">
   501→          <OEEComponentCard
   502→            label="Availability"
   503→            value={overallA}
   504→            icon={() => <span className="text-lg font-mono" style={{ color: C_GREEN }}>A</span>}
   505→            color={C_GREEN}
   506→          />
   507→          <span className="text-2xl font-bold text-muted-foreground shrink-0">×</span>
   508→          <OEEComponentCard
   509→            label="Performance"
   510→            value={overallP}
   511→            icon={() => <span className="text-lg font-mono" style={{ color: C_CYAN }}>P</span>}
   512→            color={C_CYAN}
   513→          />
   514→          <span className="text-2xl font-bold text-muted-foreground shrink-0">×</span>
   515→          <OEEComponentCard
   516→            label="Quality"
   517→            value={overallQ}
   518→            icon={() => <span className="text-lg font-mono" style={{ color: C_ORANGE }}>Q</span>}
