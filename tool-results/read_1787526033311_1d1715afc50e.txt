   188→   188→function generateApiData(): { time: string; responseTime: number }[] {
   189→   189→  const data: { time: string; responseTime: number }[] = []
   190→   190→  const now = Date.now()
   191→   191→  for (let i = 59; i >= 0; i--) {
   192→   192→    const t = (60 - i) / 60
   193→   193→    const base = 38 + 8 * Math.sin(t * Math.PI * 3)
   194→   194→    const spike = i === 12 ? 180 : i === 35 ? 95 : 0
   195→   195→    data.push({
   196→   196→      time: new Date(now - i * 5000).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' }),
   197→   197→      responseTime: Math.max(10, base + (Math.random() - 0.5) * 20 + spike),
   198→   198→    })
   199→   199→  }
   200→   200→  return data
   201→   201→}

   203→   203→function generateMemoryData(): { time: string; usage: number }[] {
   204→   204→  const data: { time: string; usage: number }[] = []
   205→   205→  const now = Date.now()
   206→   206→  let mem = 68
   207→   207→  for (let i = 59; i >= 0; i--) {
   208→   208→    mem += (Math.random() - 0.48) * 3
   209→   209→    mem = Math.max(55, Math.min(82, mem))
   210→   210→    data.push({
   211→   211→      time: new Date(now - i * 5000).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' }),
   212→   212→      usage: Math.round(mem * 10) / 10,
   213→   213→    })
   214→   214→  }
   215→   215→  return data
   216→   216→}

   219→   219→function isHealthy(status: ServiceStatus['status']): boolean {
   220→   220→  return ['healthy', 'running', 'connected', 'active'].includes(status)
   221→   221→}

   223→   223→function statusColor(status: ServiceStatus['status']): string {
   224→   224→  if (isHealthy(status)) return 'bg-emerald-500'
   225→   225→  if (status === 'degraded') return 'bg-amber-500'
   226→   226→  return 'bg-red-500'
   227→   227→}

   229→   229→function statusLabel(status: ServiceStatus['status']): string {
   230→   230→  const map: Record<string, string> = {
   231→   231→    healthy: 'Healthy',
   232→   232→    running: 'Running',
   233→   233→    connected: 'Connected',
   234→   234→    active: 'Active',
   235→   235→    degraded: 'Degraded',
   236→   236→    down: 'Down',
   237→   237→  }
   238→   238→  return map[status] ?? status
   239→   239→}
