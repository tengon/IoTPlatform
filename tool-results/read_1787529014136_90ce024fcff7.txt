   278→// ─── Custom dark tooltip ────────────────────────────────────────────────
   279→function DarkTooltip({
   280→  active,
   281→  payload,
   282→  label,
   283→}: {
   284→  active?: boolean
   285→  payload?: Array<{ name: string; value: number; color: string }>
   286→  label?: string
   287→}) {
   288→  if (!active || !payload?.length) return null
   289→  return (
   290→    <div className="chart-tooltip rounded-lg bg-card/95 border border-border/60 px-3 py-2 shadow-xl">
   291→      <p className="text-xs text-muted-foreground mb-1">{label}</p>
   292→      {payload.map((p, i) => (
   293→        <p key={i} className="text-xs font-medium" style={{ color: p.color }}>
   294→          {p.name}: {p.value.toLocaleString()}
   295→        </p>
   296→      ))}
   297→    </div>
   298→  )
   299→}

   301→// ─── Type badge color map ───────────────────────────────────────────────
   302→function getTypeBadgeVariant(type: string) {
   303→  switch (type) {
   304→    case 'Production':
   305→    case 'Utilization':
   306→      return 'default' as const
   307→    case 'Alarm':
   308→      return 'destructive' as const
   309→    case 'Energy':
   310→      return 'warning' as const
   311→    case 'OEE':
   312→      return 'default' as const
   313→    case 'Maintenance':
   314→      return 'destructive' as const
   315→    default:
   316→      return 'secondary' as const
   317→  }
   318→}

   329→  const [startDate, setStartDate] = useState('2025-06-10')
   330→  const [endDate, setEndDate] = useState('2025-06-16')
   331→  const [isCustomRange, setIsCustomRange] = useState(false)
   332→  const [generateState, setGenerateState] = useState<'idle' | 'generating' | 'complete'>('idle')
   333→  const [progress, setProgress] = useState(0)
   334→  const [previewReport, setPreviewReport] = useState<RecentReport | null>(null)
   335→
   336→  // Handle quick range selection
   337→  const handleQuickRange = useCallback((range: QuickRange) => {
   338→    setQuickRange(range)
   339→    if (range === 'Custom') {
   340→      setIsCustomRange(true)
   341→    } else {
   342→      setIsCustomRange(false)
   343→      const now = new Date('2025-06-16T14:00:00')
   344→      let start = new Date(now)
   345→      if (range === 'Last 24h') start = new Date(now.getTime() - 24 * 60 * 60 * 1000)
   346→      else if (range === 'Last 7 Days') start = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000)
   347→      else if (range === 'Last 30 Days') start = new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000)
   348→      setStartDate(start.toISOString().split('T')[0])
   349→      setEndDate(now.toISOString().split('T')[0])
   350→    }
   351→    // Reset generate state when range changes
   352→    setGenerateState('idle')
   353→    setProgress(0)
   354→  }, [])
   355→
   356→  // Simulate report generation
   357→  const handleGenerate = useCallback(() => {
   358→    if (!selectedType) return

   360→    setProgress(0)
   361→  }, [selectedType])
   362→

   364→    if (generateState !== 'generating') return

   366→      setProgress((prev) => {
   367→        const next = prev + Math.random() * 15 + 5
   368→        if (next >= 100) {
   369→          setGenerateState('complete')
   370→          return 100
   371→        }
   372→        return next
   373→      })
   374→    }, 300)
   375→    return () => clearInterval(interval)
   376→  }, [generateState])
   377→
   378→  // Compute some store-derived stats for the page header
   379→  const activeMachineCount = machines.filter((m) => m.status === 'running').length
   380→  const activeAlarmCount = alarms.filter((a) => a.status === 'active').length
   381→
   382→  return (

   384→      {/* Page Header */}
   385→      <PageHeader
   386→        title="Reports"

   394→              {activeMachineCount} machines active

   399→        }
   400→      />
   401→
   402→      {/* Report Type Selector */}
   403→      <section className="animate-slide-up">
   404→        <h2 className="text-sm font-semibold text-muted-foreground/80 uppercase tracking-wider mb-3">
   405→          Select Report Type

   407→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   408→          {REPORT_TYPES.map((report, idx) => {
   409→            const Icon = report.icon
   410→            const isSelected = selectedType === report.id
   411→            return (
   412→              <Card
   413→                key={report.id}
   414→                className={
   415→                  `kpi-card-hover cursor-pointer transition-all animate-slide-up stagger-${idx + 1} ` +
   416→                  (isSelected
   417→                    ? 'ring-1 ring-primary/40 border-primary/30 bg-primary/5'
   418→                    : 'hover:border-border/60')
   419→                }
   420→                onClick={() => {
   421→                  setSelectedType(report.id)
   422→                  setGenerateState('idle')
   423→                  setProgress(0)
   424→                }}
   425→              >
   426→                <CardHeader className="pb-3">
   427→                  <div className="flex items-start justify-between">
   428→                    <div

   437→                  </div>
