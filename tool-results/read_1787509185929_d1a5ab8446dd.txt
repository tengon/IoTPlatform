    92→    92→// ─── SVG Semi-circle Gauge ──────────────────────────────────────────────────
    93→    93→function OEEGauge({ value }: { value: number }) {
    94→    94→  const clamped = Math.max(0, Math.min(100, value))
    95→    95→  const color = clamped >= 85 ? C_GREEN : clamped >= 70 ? C_YELLOW : C_RED
    96→    96→  const radius = 80
    97→    97→  const cx = 100
    98→    98→  const cy = 95
    99→    99→  // Semi-circle: from 180° to 0° (left to right)
   100→   100→  const circumference = Math.PI * radius // half circle
   101→   101→  const filled = (clamped / 100) * circumference
   102→   102→
   103→   103→  // Build SVG arc path
   104→   104→  const describeArc = (
   105→   105→    x: number,
   106→   106→    y: number,
   107→   107→    r: number,
   108→   108→    startAngle: number,
   109→   109→    endAngle: number
   110→   110→  ) => {
   111→   111→    const start = {
   112→   112→      x: x + r * Math.cos((startAngle * Math.PI) / 180),
   113→   113→      y: y + r * Math.sin((startAngle * Math.PI) / 180),
   114→   114→    }
   115→   115→    const end = {
   116→   116→      x: x + r * Math.cos((endAngle * Math.PI) / 180),
   117→   117→      y: y + r * Math.sin((endAngle * Math.PI) / 180),
   118→   118→    }
   119→   119→    const largeArcFlag = endAngle - startAngle <= 180 ? '0' : '1'
   120→   120→    return `M ${start.x} ${start.y} A ${r} ${r} 0 ${largeArcFlag} 1 ${end.x} ${end.y}`
   121→   121→  }
   122→   122→
   123→   123→  const bgPath = describeArc(cx, cy, radius, 180, 0)
   124→   124→  const valueAngle = 180 - (clamped / 100) * 180
   125→   125→  const valuePath = describeArc(cx, cy, radius, 180, valueAngle)
   126→   126→
   127→   127→  return (
   128→   128→    <div className="flex flex-col items-center">
   129→   129→      <svg width="200" height="120" viewBox="0 0 200 120">
   130→   130→        {/* Background arc */}
   131→   131→        <path
   132→   132→          d={bgPath}
   133→   133→          fill="none"
   134→   134→          stroke="rgba(255,255,255,0.08)"
   135→   135→          strokeWidth="14"
   136→   136→          strokeLinecap="round"
   137→   137→        />
   138→   138→        {/* Value arc */}
   139→   139→        <path
   140→   140→          d={valuePath}
   141→   141→          fill="none"
   142→   142→          stroke={color}
   143→   143→          strokeWidth="14"
   144→   144→          strokeLinecap="round"
   145→   145→          style={{
   146→   146→            filter: `drop-shadow(0 0 6px ${color}60)`,
   147→   147→          }}
   148→   148→        />
   149→   149→        {/* Tick marks */}
   150→   150→        {[0, 25, 50, 75, 100].map((tick) => {
   151→   151→          const angle = 180 - (tick / 100) * 180
   152→   152→          const rad = (angle * Math.PI) / 180
   153→   153→          const innerR = radius - 22
   154→   154→          const outerR = radius - 16
   155→   155→          return (
   156→   156→            <line
   157→   157→              key={tick}
   158→   158→              x1={cx + innerR * Math.cos(rad)}
   159→   159→              y1={cy + innerR * Math.sin(rad)}
   160→   160→              x2={cx + outerR * Math.cos(rad)}
   161→   161→              y2={cy + outerR * Math.sin(rad)}
   162→   162→              stroke="rgba(255,255,255,0.2)"
   163→   163→              strokeWidth="1.5"
   164→   164→            />
   165→   165→          )
   166→   166→        })}
   167→   167→        {/* Tick labels */}
   168→   168→        {[0, 50, 100].map((tick) => {
   169→   169→          const angle = 180 - (tick / 100) * 180
   170→   170→          const rad = (angle * Math.PI) / 180
   171→   171→          const labelR = radius - 32
   172→   172→          return (
   173→   173→            <text
   174→   174→              key={tick}
   175→   175→              x={cx + labelR * Math.cos(rad)}
   176→   176→              y={cy + labelR * Math.sin(rad) + 3}
   177→   177→              textAnchor="middle"
   178→   178→              fill="#a1a1aa"
   179→   179→              fontSize="9"
   180→   180→            >
   181→   181→              {tick}%
   182→   182→            </text>
   183→   183→          )
   184→   184→        })}
   185→   185→        {/* Center value */}
   186→   186→        <text x={cx} y={cy - 15} textAnchor="middle" fill={color} fontSize="32" fontWeight="bold">
   187→   187→          {clamped.toFixed(1)}%
   188→   188→        </text>
   189→   189→        <text x={cx} y={cy + 5} textAnchor="middle" fill="#a1a1aa" fontSize="11">
   190→   190→          Overall OEE
   191→   191→        </text>
   192→   192→      </svg>
   193→   193→    </div>
   194→   194→  )
   195→   195→}
