   188→function generateApiData(): { time: string; responseTime: number }[] {
   189→  const data: { time: string; responseTime: number }[] = []
   190→  const now = Date.now()
   191→  for (let i = 59; i >= 0; i--) {
   192→    const t = (60 - i) / 60
   193→    const base = 38 + 8 * Math.sin(t * Math.PI * 3)
   194→    const spike = i === 12 ? 180 : i === 35 ? 95 : 0
   195→    data.push({
   196→      time: new Date(now - i * 5000).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' }),
   197→      responseTime: Math.max(10, base + (Math.random() - 0.5) * 20 + spike),
   198→    })
   199→  }
   200→  return data
   201→}
   202→
   203→function generateMemoryData(): { time: string; usage: number }[] {
   204→  const data: { time: string; usage: number }[] = []
   205→  const now = Date.now()
   206→  let mem = 68
   207→  for (let i = 59; i >= 0; i--) {
   208→    mem += (Math.random() - 0.48) * 3
   209→    mem = Math.max(55, Math.min(82, mem))
   210→    data.push({
   211→      time: new Date(now - i * 5000).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' }),
   212→      usage: Math.round(mem * 10) / 10,
   213→    })
   214→  }
   215→  return data
   216→}

   218→// ─── Status Helpers ─────────────────────────────────────────────────────
   219→function isHealthy(status: ServiceStatus['status']): boolean {
   220→  return ['healthy', 'running', 'connected', 'active'].includes(status)
   221→}
   222→
   223→function statusColor(status: ServiceStatus['status']): string {
   224→  if (isHealthy(status)) return 'bg-emerald-500'
   225→  if (status === 'degraded') return 'bg-amber-500'
   226→  return 'bg-red-500'
   227→}
   228→
   229→function statusLabel(status: ServiceStatus['status']): string {
   230→  const map: Record<string, string> = {
   231→    healthy: 'Healthy',
   232→    running: 'Running',
   233→    connected: 'Connected',
   234→    active: 'Active',
   235→    degraded: 'Degraded',
   236→    down: 'Down',
   237→  }
   238→  return map[status] ?? status
   239→}

   241→function eventTypeConfig(type: SystemEvent['type']) {
   242→  switch (type) {
   243→    case 'deployment': return { icon: Upload, color: 'text-cyan-400', bg: 'bg-cyan-500/10', label: 'Deploy' }

   843→// ─── Resource Bar ────────────────────────────────────────────────────────
   844→function ResourceBar({ metric }: { metric: ResourceMetric }) {
   845→  const percentage = (metric.value / metric.max) * 100
   846→  const isHigh = percentage > 80
   847→  const isMedium = percentage > 60
   848→
   849→  const barColor = isHigh
   850→    ? 'bg-red-500'
   851→    : isMedium
   852→      ? 'bg-amber-500'
   853→      : metric.color === C_GREEN
   854→        ? 'bg-emerald-500'
   855→        : metric.color === C_CYAN
   856→          ? 'bg-cyan-500'
   857→          : 'bg-orange-500'
   858→
   859→  const iconMap: Record<string, React.ElementType> = {
   860→    CPU: Cpu,
   861→    Memory: Server,
   862→    Disk: HardDrive,
   863→    Network: Network,
   864→  }
   865→  const ResIcon = iconMap[metric.label] || Activity
   866→
   867→  return (
   868→    <div className="space-y-3">
   869→      <div className="flex items-center justify-between">
   870→        <div className="flex items-center gap-2">
   871→          <ResIcon className="size-4 text-muted-foreground/70" />
   872→          <span className="text-xs font-semibold text-foreground/80">{metric.label}</span>
   873→        </div>
   874→        <div className="flex items-center gap-1.5">
   875→          {metric.trend === 'up' && <ArrowUpRight className="size-3 text-amber-400" />}
   876→          {metric.trend === 'down' && <ArrowDownRight className="size-3 text-emerald-400" />}
   877→          {metric.trend === 'stable' && <Minus className="size-3 text-muted-foreground/50" />}
   878→          <span className={`text-sm font-bold metric-value ${isHigh ? 'text-red-400' : isMedium ? 'text-amber-400' : 'text-foreground'}`}>
   879→            {Math.round(metric.value)}
   880→            <span className="text-[10px] font-normal text-muted-foreground/60 ml-0.5">
   881→              {metric.unit}
   882→            </span>
   883→          </span>
   884→        </div>
   885→      </div>
   886→      <div className="relative h-2.5 w-full rounded-full bg-muted/50 overflow-hidden">
   887→        <div
   888→          className={`h-full rounded-full transition-all duration-700 ease-out ${barColor}`}
   889→          style={{
   890→            width: `${Math.min(percentage, 100)}%`,
   891→            boxShadow: isHigh
   892→              ? '0 0 12px rgba(239,68,68,0.3)'
   893→              : isMedium
   894→                ? '0 0 8px rgba(245,158,11,0.2)'
   895→                : `0 0 8px ${metric.color}33`,
   896→          }}
   897→        />
   898→        {/* 60% and 80% markers */}
   899→        {metric.label !== 'Network' && (
   900→          <>
   901→            <div
   902→              className="absolute top-0 bottom-0 w-px bg-cyan-500/30"
   903→              style={{ left: '60%' }}
   904→            />
   905→            <div
   906→              className="absolute top-0 bottom-0 w-px bg-orange-500/30"
   907→              style={{ left: '80%' }}
   908→            />
   909→          </>
   910→        )}
   911→      </div>
   912→    </div>
   913→  )
   914→}
