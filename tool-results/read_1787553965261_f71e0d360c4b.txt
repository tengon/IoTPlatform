   214→function generateApiData(): { time: string; responseTime: number }[] {
   215→  const data: { time: string; responseTime: number }[] = []
   216→  const now = Date.now()
   217→  for (let i = 59; i >= 0; i--) {
   218→    const t = (60 - i) / 60
   219→    const base = 38 + 8 * Math.sin(t * Math.PI * 3)
   220→    const spike = i === 12 ? 180 : i === 35 ? 95 : 0
   221→    data.push({
   222→      time: new Date(now - i * 5000).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' }),
   223→      responseTime: Math.max(10, base + (Math.random() - 0.5) * 20 + spike),
   224→    })
   225→  }
   226→  return data
   227→}

   229→function generateMemoryData(): { time: string; usage: number }[] {
   230→  const data: { time: string; usage: number }[] = []
   231→  const now = Date.now()
   232→  let mem = 68
   233→  for (let i = 59; i >= 0; i--) {
   234→    mem += (Math.random() - 0.48) * 3
   235→    mem = Math.max(55, Math.min(82, mem))
   236→    data.push({
   237→      time: new Date(now - i * 5000).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' }),
   238→      usage: Math.round(mem * 10) / 10,
   239→    })
   240→  }
   241→  return data
   242→}
