   106→   106→// ─── TinySparkline Component ─────────────────────────────────────────
   107→   107→
   108→   108→function TinySparkline({ data, color, width = 60, height = 20 }: { data: number[]; color: string; width?: number; height?: number }) {
   109→   109→  if (data.length < 2) return null
   110→   110→
   111→   111→  const min = Math.min(...data)
   112→   112→  const max = Math.max(...data)
   113→   113→  const range = max - min || 1
   114→   114→  const padding = 1
   115→   115→
   116→   116→  const points = data.map((v, i) => {
   117→   117→    const x = padding + (i / (data.length - 1)) * (width - padding * 2)
   118→   118→    const y = height - padding - ((v - min) / range) * (height - padding * 2)
   119→   119→    return `${x},${y}`
   120→   120→  })
   121→   121→
   122→   122→  const lastX = padding + ((data.length - 1) / (data.length - 1)) * (width - padding * 2)
   123→   123→  const lastY = height - padding - ((data[data.length - 1] - min) / range) * (height - padding * 2)
   124→   124→
   125→   125→  const gradientId = `spark-${color.replace('#', '')}`
   126→   126→
   127→   127→  return (
   128→   128→    <svg viewBox={`0 0 ${width} ${height}`} className="inline-block ml-1.5 align-middle">
   129→   129→      <defs>
   130→   130→        <linearGradient id={gradientId} x1="0" y1="0" x2="0" y2="1">
   131→   131→          <stop offset="0%" stopColor={color} stopOpacity={0.15} />
   132→   132→          <stop offset="100%" stopColor={color} stopOpacity={0} />
   133→   133→        </linearGradient>
   134→   134→      </defs>
   135→   135→      <polyline
   136→   136→        points={points.join(' ')}
   137→   137→        stroke={color}
   138→   138→        strokeWidth={1.5}
   139→   139→        fill="none"
   140→   140→        strokeLinecap="round"
   141→   141→        strokeLinejoin="round"
   142→   142→      />
   143→   143→      <polygon
   144→   144→        points={`${points.join(' ')} ${lastX},${height} ${padding},${height}`}
   145→   145→        fill={`url(#${gradientId})`}
   146→   146→      />
   147→   147→      <circle cx={lastX} cy={lastY} r={2} fill={color} />
   148→   148→    </svg>
   149→   149→  )
   150→   150→}

   166→   166→// ─── Trend Arrow ─────────────────────────────────────────────────────
   167→   167→
   168→   168→function TrendArrow({ value, prev }: { value: number; prev: number | undefined }) {
   169→   169→  if (prev === undefined) return <Minus className="size-3 text-muted-foreground" />
   170→   170→  const diff = value - prev
   171→   171→  if (Math.abs(diff) < 0.5) return <Minus className="size-3 text-muted-foreground" />
   172→   172→  if (diff > 0) return <TrendingUp className="size-3 text-emerald-400" />
   173→   173→  return <TrendingDown className="size-3 text-red-400" />
   174→   174→}

   365→   365→export function LiveMonitoringPage() {

   370→   370→  const liveTelemetry = useIIoTStore((s) => s.liveTelemetry)
   371→   371→
   372→   372→  // Force re-render every 2 seconds to pick up store changes
   373→   373→  const [tick, setTick] = useState(0)
   374→   374→  const [lastUpdatedText, setLastUpdatedText] = useState('—')
   375→   375→  const [selectedMachine, setSelectedMachine] = useState<MachineStatus | null>(null)
   376→   376→  const [detailOpen, setDetailOpen] = useState(false)
   377→   377→
   378→   378→  useEffect(() => {
   379→   379→    const interval = setInterval(() => setTick((k) => k + 1), 2000)
   380→   380→    return () => clearInterval(interval)
   381→   381→  }, [])
   382→   382→
   383→   383→  // Update last-updated text
   384→   384→  useEffect(() => {
   385→   385→    function update() {
   386→   386→      if (lastUpdate) {
   387→   387→        setLastUpdatedText(formatDistanceToNow(new Date(lastUpdate), { addSuffix: true }))
   388→   388→      }
   389→   389→    }
   390→   390→    update()
   391→   391→    const interval = setInterval(update, 10000)
   392→   392→    return () => clearInterval(interval)
   393→   393→  }, [lastUpdate])
   394→   394→
   395→   395→  // Flatten all live telemetry into a displayable list
   396→   396→  const flatTelemetry = useMemo(() => {
   397→   397→    const entries: { key: string; label: string; value: number; unit?: string }[] = []
   398→   398→    // Device-level telemetry from WebSocket (temperature data)
   399→   399→    Object.entries(liveTelemetry).forEach(([id, points]) => {
   400→   400→      if (points.length > 0) {
   401→   401→        const latest = points[points.length - 1]
   402→   402→        const machine = machines.find((m) => m.id === id)
   403→   403→        const device = devices.find((d) => d.id === id)
   404→   404→        const label = machine?.name || device?.name || id
   405→   405→        entries.push({
   406→   406→          key: `${id}-temp`,
   407→   407→          label: `${label} (Temp)`,
   408→   408→          value: latest.value,
   409→   409→          unit: '°C',
   410→   410→        })
   411→   411→      }
   412→   412→    })
   413→   413→    // Machine-level metrics from store
   414→   414→    machines.forEach((m) => {
   415→   415→      entries.push(
   416→   416→        { key: `${m.id}-rpm`, label: `${m.name} (RPM)`, value: m.rpm, unit: 'rpm' },
   417→   417→        { key: `${m.id}-power`, label: `${m.name} (Power)`, value: m.power, unit: 'kW' },
   418→   418→        { key: `${m.id}-oee`, label: `${m.name} (OEE)`, value: m.oee, unit: '%' },
   419→   419→      )
   420→   420→    })
   421→   421→    return entries
   422→   422→  }, [liveTelemetry, machines, devices])
   423→   423→
   424→   424→  return (
   425→   425→    <div className="space-y-6">
   426→   426→      <PageHeader
   427→   427→        icon={Activity}
   428→   428→        title="Live Monitoring"
   429→   429→        description="Real-time machine and device telemetry"
   430→   430→        lastUpdated={lastUpdatedText}
   431→   431→        badge={isConnected ? 'STREAMING' : 'OFFLINE'}
   432→   432→      />
   433→   433→
   434→   434→      {/* Connection Status Bar */}
