   112→// ─── Mini sparkline for KPI cards ─────────────────────────────────────────────
   113→function MiniSparkline({ data, color }: { data: number[]; color: string }) {
   114→  const sparkData = data.slice(-12).map((v, i) => ({ i, v }))
   115→  if (sparkData.length < 2) return null
   116→  const min = Math.min(...sparkData.map((d) => d.v))
   117→  const max = Math.max(...sparkData.map((d) => d.v))
   118→  const range = max - min || 1
   119→  const h = 28
   120→  const w = 72
   121→  const step = w / (sparkData.length - 1)
   122→  const points = sparkData
   123→    .map((d, idx) => {
   124→      const x = idx * step
   125→      const y = h - ((d.v - min) / range) * (h - 4) - 2
   126→      return `${x},${y}`
   127→    })
   128→    .join(' ')
   129→  return (
   130→    <svg viewBox={`0 0 ${w} ${h}`} className="w-[72px] h-[28px] opacity-60">
   131→      <polyline
   132→        fill="none"
   133→        stroke={color}
   134→        strokeWidth="1.5"
   135→        strokeLinejoin="round"
   136→        strokeLinecap="round"
   137→        points={points}
   138→      />
   139→    </svg>
   140→  )
   141→}
