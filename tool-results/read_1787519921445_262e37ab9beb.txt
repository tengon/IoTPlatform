   106→// ─── TinySparkline Component ─────────────────────────────────────────
   107→
   108→function TinySparkline({ data, color, width = 60, height = 20 }: { data: number[]; color: string; width?: number; height?: number }) {
   109→  if (data.length < 2) return null
   110→
   111→  const min = Math.min(...data)
   112→  const max = Math.max(...data)
   113→  const range = max - min || 1
   114→  const padding = 1
   115→
   116→  const points = data.map((v, i) => {
   117→    const x = padding + (i / (data.length - 1)) * (width - padding * 2)
   118→    const y = height - padding - ((v - min) / range) * (height - padding * 2)
   119→    return `${x},${y}`
   120→  })
   121→
   122→  const lastX = padding + ((data.length - 1) / (data.length - 1)) * (width - padding * 2)
   123→  const lastY = height - padding - ((data[data.length - 1] - min) / range) * (height - padding * 2)
   124→
   125→  const gradientId = `spark-${color.replace('#', '')}`
   126→
   127→  return (
   128→    <svg viewBox={`0 0 ${width} ${height}`} className="inline-block ml-1.5 align-middle">
   129→      <defs>
   130→        <linearGradient id={gradientId} x1="0" y1="0" x2="0" y2="1">
   131→          <stop offset="0%" stopColor={color} stopOpacity={0.15} />
   132→          <stop offset="100%" stopColor={color} stopOpacity={0} />
   133→        </linearGradient>
   134→      </defs>
   135→      <polyline
   136→        points={points.join(' ')}
   137→        stroke={color}
   138→        strokeWidth={1.5}
   139→        fill="none"
   140→        strokeLinecap="round"
   141→        strokeLinejoin="round"
   142→      />
   143→      <polygon
   144→        points={`${points.join(' ')} ${lastX},${height} ${padding},${height}`}
   145→        fill={`url(#${gradientId})`}
   146→      />
   147→      <circle cx={lastX} cy={lastY} r={2} fill={color} />
   148→    </svg>
   149→  )
   150→}

   166→// ─── Trend Arrow ─────────────────────────────────────────────────────
   167→
   168→function TrendArrow({ value, prev }: { value: number; prev: number | undefined }) {
   169→  if (prev === undefined) return <Minus className="size-3 text-muted-foreground" />
   170→  const diff = value - prev
   171→  if (Math.abs(diff) < 0.5) return <Minus className="size-3 text-muted-foreground" />
   172→  if (diff > 0) return <TrendingUp className="size-3 text-emerald-400" />
   173→  return <TrendingDown className="size-3 text-red-400" />
   174→}

   190→        <Line
   191→          type="monotone"

   369→  const devices = useIIoTStore((s) => s.devices)
   370→  const liveTelemetry = useIIoTStore((s) => s.liveTelemetry)
   371→
   372→  // Force re-render every 2 seconds to pick up store changes
   373→  const [tick, setTick] = useState(0)
   374→  const [lastUpdatedText, setLastUpdatedText] = useState('—')
   375→  const [selectedMachine, setSelectedMachine] = useState<MachineStatus | null>(null)
   376→  const [detailOpen, setDetailOpen] = useState(false)
   377→
   378→  useEffect(() => {
   379→    const interval = setInterval(() => setTick((k) => k + 1), 2000)
   380→    return () => clearInterval(interval)
   381→  }, [])
   382→
   383→  // Update last-updated text
   384→  useEffect(() => {
   385→    function update() {
   386→      if (lastUpdate) {
   387→        setLastUpdatedText(formatDistanceToNow(new Date(lastUpdate), { addSuffix: true }))
   388→      }
   389→    }
   390→    update()
   391→    const interval = setInterval(update, 10000)
   392→    return () => clearInterval(interval)

   405→          key: `${id}-temp`,

   414→        { key: `${m.id}-rpm`, label: `${m.name} (RPM)`, value: m.rpm, unit: 'rpm' },

   416→        { key: `${m.id}-oee`, label: `${m.name} (OEE)`, value: m.oee, unit: '%' },
   417→      )
   418→    })

   420→  }, [liveTelemetry, machines, devices])

   434→        isConnected

   441→          <WifiOff className="size-4 text-red-400" />
   442→        )}
   443→        <div className={`size-2 rounded-full ${isConnected ? 'bg-emerald-500 animate-pulse-dot' : 'bg-red-500'}`} />
   444→        <span className="text-sm font-medium">
   445→          {isConnected ? 'Connected to IIoT Gateway' : 'Disconnected — Reconnecting...'}

   448→          <span className="ml-auto text-xs text-muted-foreground/60 flex items-center gap-1.5">
   449→            <span className="size-1 rounded-full bg-emerald-500 animate-pulse" />

   453→          <AlertTriangle className="ml-auto size-4 text-amber-400" />
