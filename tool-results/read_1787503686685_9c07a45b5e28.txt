   120→   120→  const step = w / (sparkData.length - 1)
   121→   121→  const points = sparkData
   122→   122→    .map((d, idx) => {
   123→   123→      const x = idx * step
   124→   124→      const y = h - ((d.v - min) / range) * (h - 4) - 2
   125→   125→      return `${x},${y}`
   126→   126→    })
   127→   127→    .join(' ')
   128→   128→  return (
   129→   129→    <svg viewBox={`0 0 ${w} ${h}`} className="w-[72px] h-[28px] opacity-60">
   130→   130→      <polyline
   131→   131→        fill="none"
   132→   132→        stroke={color}
   133→   133→        strokeWidth="1.5"
   134→   134→        strokeLinejoin="round"
   135→   135→        strokeLinecap="round"
   136→   136→        points={points}
   137→   137→      />
   138→   138→    </svg>
   139→   139→  )
   140→   140→}
