   260→interface RolePermission {
   261→  id: string
   262→  name: string
   263→  description: string
   264→  userCount: number
   265→  color: string
   266→  bg: string
   267→  border: string
   268→  permissions: {
   269→    Dashboard: boolean
   270→    Monitoring: boolean
   271→    Analytics: boolean
   272→    Alerts: boolean
   273→    Devices: boolean
   274→    Admin: boolean
   275→  }
   276→}
   277→
   278→const permissionKeys = ['Dashboard', 'Monitoring', 'Analytics', 'Alerts', 'Devices', 'Admin'] as const
   279→
   280→const initialRoles: RolePermission[] = [
   281→  { id: 'role-admin', name: 'Admin', description: 'Full system access with all administrative privileges', userCount: 2, color: 'text-red-400', bg: 'bg-red-500/10', border: 'border-red-500/30', permissions: { Dashboard: true, Monitoring: true, Analytics: true, Alerts: true, Devices: true, Admin: true } },
   282→  { id: 'role-manager', name: 'Manager', description: 'Supervisory access with production and analytics capabilities', userCount: 3, color: 'text-amber-400', bg: 'bg-amber-500/10', border: 'border-amber-500/30', permissions: { Dashboard: true, Monitoring: true, Analytics: true, Alerts: true, Devices: true, Admin: false } },
   283→  { id: 'role-supervisor', name: 'Supervisor', description: 'Shift oversight with monitoring and alert management', userCount: 5, color: 'text-cyan-400', bg: 'bg-cyan-500/10', border: 'border-cyan-500/30', permissions: { Dashboard: true, Monitoring: true, Analytics: true, Alerts: true, Devices: false, Admin: false } },
   284→  { id: 'role-engineer', name: 'Engineer', description: 'Technical access for device management and analytics', userCount: 8, color: 'text-emerald-400', bg: 'bg-emerald-500/10', border: 'border-emerald-500/30', permissions: { Dashboard: true, Monitoring: true, Analytics: true, Alerts: true, Devices: true, Admin: false } },
   285→  { id: 'role-operator', name: 'Operator', description: 'Basic access for monitoring and dashboard viewing', userCount: 15, color: 'text-muted-foreground', bg: 'bg-primary/10', border: 'border-primary/30', permissions: { Dashboard: true, Monitoring: true, Analytics: false, Alerts: true, Devices: false, Admin: false } },
   286→]
   287→
   288→export function RolesPermissionsPage() {
   289→  const [roles, setRoles] = useState<RolePermission[]>(initialRoles)
   290→  const [editingRole, setEditingRole] = useState<RolePermission | null>(null)
   291→  const [tempPerms, setTempPerms] = useState<RolePermission['permissions'] | null>(null)
   292→  const lastUpdate = useIIoTStore((s) => s.lastUpdate)
   293→  const [lastUpdatedText, setLastUpdatedText] = useState('')
   294→
   295→  useEffect(() => {
   296→    function update() {
   297→      if (lastUpdate) {
   298→        setLastUpdatedText(formatDistanceToNow(new Date(lastUpdate), { addSuffix: true }))
   299→      }
   300→    }
   301→    update()
   302→    const interval = setInterval(update, 10000)
   303→    return () => clearInterval(interval)
   304→  }, [lastUpdate])
   305→
   306→  const openEditDialog = (role: RolePermission) => {
   307→    setEditingRole(role)
   308→    setTempPerms({ ...role.permissions })
   309→  }
   310→
   311→  const savePermissions = () => {
   312→    if (!editingRole || !tempPerms) return
   313→    setRoles(roles.map((r) => r.id === editingRole.id ? { ...r, permissions: tempPerms } : r))
   314→    setEditingRole(null)
   315→    setTempPerms(null)
   316→  }
   317→
   318→  const togglePerm = (key: keyof RolePermission['permissions']) => {
   319→    if (!tempPerms) return
   320→    setTempPerms({ ...tempPerms, [key]: !tempPerms[key] })
   321→  }
   322→
   323→  return (
   324→    <div className="space-y-6 animate-slide-up">
   325→      <PageHeader
   326→        icon={Shield}
   327→        title="Roles & Permissions"
   328→        description="Configure user roles and permissions"
   329→        lastUpdated={lastUpdatedText}
   330→      />
   331→
   332→      <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
   333→        {roles.map((role, idx) => (
   334→          <Card key={role.id} className="border-border/40 hover:border-border/60 transition-colors duration-300" style={{ animationDelay: `${idx * 50}ms` }}>
   335→            <CardHeader className="pb-3 pt-5 px-5">
   336→              <div className="flex items-start justify-between">
   337→                <div className="space-y-1">
   338→                  <div className="flex items-center gap-2">
   339→                    <CardTitle className="text-base font-semibold">{role.name}</CardTitle>
   340→                    <Badge variant="outline" className={`${role.bg} ${role.color} ${role.border} text-xs`}>{role.userCount} users</Badge>
   341→                  </div>
   342→                  <CardDescription className="text-xs">{role.description}</CardDescription>
   343→                </div>
   344→                <Button variant="ghost" size="sm" className="h-8 w-8 p-0 text-muted-foreground hover:text-foreground hover:bg-muted/50" onClick={() => openEditDialog(role)}>
   345→                  <Pencil className="size-3.5" />
   346→                </Button>
   347→              </div>
   348→            </CardHeader>
   349→            <CardContent className="px-5 pb-5">
   350→              <Separator className="mb-3" />
   351→              <div className="grid grid-cols-2 gap-x-4 gap-y-2.5">
   352→                {permissionKeys.map((key) => (
   353→                  <div key={key} className="flex items-center gap-2">
   354→                    <div className={`size-4 rounded flex items-center justify-center shrink-0 ${role.permissions[key] ? 'bg-emerald-500/20' : 'bg-muted/50'}`}>
   355→                      {role.permissions[key] && (
   356→                        <svg className="size-3 text-emerald-400" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={3}>
   357→                          <path strokeLinecap="round" strokeLinejoin="round" d="M5 13l4 4L19 7" />
   358→                        </svg>
   359→                      )}
   360→                    </div>
   361→                    <span className={`text-xs ${role.permissions[key] ? 'text-foreground' : 'text-muted-foreground/60'}`}>{key}</span>
   362→                  </div>
   363→                ))}
   364→              </div>
   365→            </CardContent>
   366→          </Card>
   367→        ))}
   368→      </div>
   369→
   370→      <Dialog open={!!editingRole} onOpenChange={(open) => { if (!open) { setEditingRole(null); setTempPerms(null) } }}>
   371→        <DialogContent className="sm:max-w-[440px]">
   372→          <DialogHeader>
   373→            <DialogTitle>Edit Permissions - {editingRole?.name}</DialogTitle>
   374→            <DialogDescription>Toggle permissions for the {editingRole?.name} role</DialogDescription>
   375→          </DialogHeader>
   376→          <Separator />
   377→          <div className="py-2 space-y-3">
   378→            {permissionKeys.map((key) => (
   379→              <div key={key} className="flex items-center justify-between py-1">
   380→                <Label className="text-sm">{key}</Label>
   381→                <Switch
   382→                  checked={tempPerms?.[key] ?? false}
   383→                  onCheckedChange={() => togglePerm(key)}
   384→                />
   385→              </div>
   386→            ))}
   387→          </div>
   388→          <Separator />
   389→          <DialogFooter>
   390→            <Button variant="outline" onClick={() => { setEditingRole(null); setTempPerms(null) }}>Cancel</Button>
   391→            <Button onClick={savePermissions}>Save Permissions</Button>
   392→          </DialogFooter>
   393→        </DialogContent>
   394→      </Dialog>
   395→    </div>
   396→  )
   397→}
