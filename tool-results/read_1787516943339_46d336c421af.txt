   113→// ─── SVG Semi-circle Gauge (enhanced with gradient + glow) ──────────────────
   114→function OEEGauge({ value }: { value: number }) {
   115→  const clamped = Math.max(0, Math.min(100, value))
   116→  const color = clamped >= 85 ? C_GREEN : clamped >= 70 ? C_YELLOW : C_RED
   117→  const radius = 80
   118→  const cx = 100
   119→  const cy = 95
   120→  // Semi-circle: from 180° to 0° (left to right)
   121→  const circumference = Math.PI * radius // half circle
   122→  const filled = (clamped / 100) * circumference
   123→
   124→  const gradId = `oee-gauge-grad-${color.replace('#', '')}`
   125→  const glowId = `oee-gauge-glow-${color.replace('#', '')}`
   126→
   127→  // Build SVG arc path
   128→  const describeArc = (
   129→    x: number,
   130→    y: number,
   131→    r: number,
   132→    startAngle: number,
   133→    endAngle: number
   134→  ) => {
   135→    const start = {
   136→      x: x + r * Math.cos((startAngle * Math.PI) / 180),
   137→      y: y + r * Math.sin((startAngle * Math.PI) / 180),
   138→    }
   139→    const end = {
   140→      x: x + r * Math.cos((endAngle * Math.PI) / 180),
   141→      y: y + r * Math.sin((endAngle * Math.PI) / 180),
   142→    }
   143→    const largeArcFlag = endAngle - startAngle <= 180 ? '0' : '1'
   144→    return `M ${start.x} ${start.y} A ${r} ${r} 0 ${largeArcFlag} 1 ${end.x} ${end.y}`
   145→  }
   146→
   147→  const bgPath = describeArc(cx, cy, radius, 180, 0)
   148→  const valueAngle = 180 - (clamped / 100) * 180
   149→  const valuePath = describeArc(cx, cy, radius, 180, valueAngle)
   150→
   151→  return (
   152→    <div className="flex flex-col items-center">
   153→      <svg width="200" height="120" viewBox="0 0 200 120">
   154→        <defs>
   155→          <linearGradient id={gradId} x1="0%" y1="0%" x2="100%" y2="0%">
   156→            <stop offset="0%" stopColor={color} stopOpacity="0.6" />
   157→            <stop offset="50%" stopColor={color} stopOpacity="1" />
   158→            <stop offset="100%" stopColor={color} stopOpacity="0.7" />
   159→          </linearGradient>
   160→          <filter id={glowId}>
   161→            <feGaussianBlur stdDeviation="4" result="blur" />
   162→            <feFlood floodColor={color} floodOpacity="0.4" result="color" />
   163→            <feComposite in="color" in2="blur" operator="in" result="glow" />
   164→            <feMerge>
   165→              <feMergeNode in="glow" />
   166→              <feMergeNode in="SourceGraphic" />
   167→            </feMerge>
   168→          </filter>
   169→        </defs>
   170→        {/* Background arc */}
   171→        <path
   172→          d={bgPath}
   173→          fill="none"
   174→          stroke="rgba(255,255,255,0.06)"
   175→          strokeWidth="14"
   176→          strokeLinecap="round"
   177→        />
   178→        {/* Value arc with gradient + glow */}
   179→        <path
   180→          d={valuePath}
   181→          fill="none"
   182→          stroke={`url(#${gradId})`}
   183→          strokeWidth="14"
   184→          strokeLinecap="round"
   185→          filter={`url(#${glowId})`}
   186→        />
   187→        {/* Tick marks */}
   188→        {[0, 25, 50, 75, 100].map((tick) => {
   189→          const angle = 180 - (tick / 100) * 180
   190→          const rad = (angle * Math.PI) / 180
   191→          const innerR = radius - 22
   192→          const outerR = radius - 16
   193→          return (
   194→            <line
   195→              key={tick}
   196→              x1={cx + innerR * Math.cos(rad)}
   197→              y1={cy + innerR * Math.sin(rad)}
   198→              x2={cx + outerR * Math.cos(rad)}
   199→              y2={cy + outerR * Math.sin(rad)}
   200→              stroke="rgba(255,255,255,0.2)"
   201→              strokeWidth="1.5"
   202→            />
   203→          )
   204→        })}
   205→        {/* Tick labels */}
   206→        {[0, 50, 100].map((tick) => {
   207→          const angle = 180 - (tick / 100) * 180
   208→          const rad = (angle * Math.PI) / 180
   209→          const labelR = radius - 32
   210→          return (
   211→            <text
   212→              key={tick}
   213→              x={cx + labelR * Math.cos(rad)}
   214→              y={cy + labelR * Math.sin(rad) + 3}
   215→              textAnchor="middle"
   216→              fill="rgba(255,255,255,0.35)"
   217→              fontSize="9"
   218→            >
   219→              {tick}%
   220→            </text>
   221→          )
   222→        })}
   223→        {/* Center value */}
   224→        <text x={cx} y={cy - 15} textAnchor="middle" fill={color} fontSize="32" fontWeight="bold">
   225→          {clamped.toFixed(1)}%
   226→        </text>
   227→        <text x={cx} y={cy + 5} textAnchor="middle" fill="rgba(255,255,255,0.4)" fontSize="11">
   228→          Overall OEE
   229→        </text>
   230→      </svg>
   231→    </div>
   232→  )
   233→}

   279→export function OEEPage() {
   280→  const { machines, lastUpdate } = useIIoTStore()
   281→  const [lastUpdatedText, setLastUpdatedText] = useState('—')
   282→
   283→  useEffect(() => {
   284→    function update() {
   285→      if (lastUpdate) {
   286→        setLastUpdatedText(formatDistanceToNow(new Date(lastUpdate), { addSuffix: true }))
   287→      }
   288→    }
   289→    update()
   290→    const interval = setInterval(update, 10000)
   291→    return () => clearInterval(interval)
   292→  }, [lastUpdate])
   293→
   294→  const machineData = useMemo(
   295→    () => (machines.length > 0 ? machines : DEFAULT_MACHINES),
   296→    [machines]
   297→  )
   298→
   299→  // ── Overall OEE ───────────────────────────────────────────────────────────
   300→  const overallOEE = useMemo(
   301→    () => machineData.reduce((sum, m) => sum + m.oee, 0) / machineData.length,
   302→    [machineData]
   303→  )
   304→
   305→  const overallA = useMemo(
   306→    () => machineData.reduce((sum, m) => sum + m.availability, 0) / machineData.length,
   307→    [machineData]
   308→  )
   309→
   310→  const overallP = useMemo(
   311→    () => machineData.reduce((sum, m) => sum + m.performance, 0) / machineData.length,
   312→    [machineData]
   313→  )
   314→
   315→  const overallQ = useMemo(
   316→    () => machineData.reduce((sum, m) => sum + m.quality, 0) / machineData.length,
   317→    [machineData]
   318→  )
   319→
   320→  // ── 24h OEE trend data ──────────────────────────────────────────────────
   321→  const oeeTrend24h = useMemo(() => {
   322→    const data: Array<{ hour: string; OEE: number }> = []
   323→    let val = overallOEE
   324→    for (let i = 0; i < 24; i++) {
   325→      val = Math.max(65, Math.min(95, val + (Math.random() - 0.45) * 6))
   326→      data.push({
   327→        hour: `${String(i).padStart(2, '0')}:00`,
   328→        OEE: Math.round(val * 10) / 10,
   329→      })
   330→    }
   331→    return data
   332→  }, [overallOEE])
   333→
   334→  // ── 30-day OEE trend data ─────────────────────────────────────────────────
   335→  const trendData = useMemo(() => {
   336→    const data: Array<{
   337→      day: string
   338→      Availability: number
   339→      Performance: number
   340→      Quality: number
   341→    }> = []
   342→    for (let i = 29; i >= 0; i--) {
   343→      data.push({
   344→        day: `D-${i}`,
   345→        Availability: Math.min(100, overallA + (Math.random() - 0.5) * 6),
   346→        Performance: Math.min(100, overallP + (Math.random() - 0.5) * 4),
   347→        Quality: Math.min(100, overallQ + (Math.random() - 0.5) * 1.5),
   348→      })
   349→    }
   350→    return data
   351→  }, [overallA, overallP, overallQ])
   352→
   353→  // ── Loss analysis data ────────────────────────────────────────────────────
   354→  const lossData = useMemo(
   355→    () =>
   356→      machineData.map((m) => ({
   357→        name: m.name.replace(/ #\d+/, ''),
   358→        'Avail. Loss': Math.round((100 - m.availability) * 10) / 10,
   359→        'Perf. Loss': Math.round((100 - m.performance) * 10) / 10,
   360→        'Quality Loss': Math.round((100 - m.quality) * 10) / 10,
   361→      })),
   362→    [machineData]
   363→  )
   364→
   365→  // ── Target vs Actual ──────────────────────────────────────────────────────
   366→  const targetVsActual = useMemo(
   367→    () =>
   368→      machineData.map((m) => ({
   369→        name: m.name.replace(/ #\d+/, ''),
   370→        Target: OEE_TARGET,
   371→        Actual: m.oee,
   372→      })),
   373→    [machineData]
   374→  )
   375→
   376→  return (
   377→    <div className="space-y-6">
   378→      <PageHeader
   379→        icon={Gauge}
   380→        title="OEE Analysis"
   381→        description="Overall Equipment Effectiveness breakdown"
   382→        lastUpdated={lastUpdatedText}
   383→      />
   384→
   385→      {/* OEE Gauge */}
   386→      <Card className="chart-container-glass hover:border-border/60 transition-colors duration-300 animate-slide-up">
   387→        <CardContent className="pt-6 flex flex-col items-center px-5 pb-5">
   388→          <OEEGauge value={overallOEE} />
   389→          <div className="flex items-center gap-2 mt-2">
   390→            <Target className="h-3.5 w-3.5 text-muted-foreground" />
   391→            <span className="text-xs text-muted-foreground">
   392→              Target: {OEE_TARGET}%{' '}
   393→              {overallOEE >= OEE_TARGET ? (
   394→                <span className="text-emerald-400 font-medium">
   395→                  <ArrowUpRight className="inline h-3 w-3" /> On Track

   407→      {/* OEE Trend (24h) */}
